One order service, end to end

A POST /orders that survives a client retry, a failover mid-transaction and a consumer that was down for an hour. Every hop below is a service from an earlier section, wired in the order that makes those three things true.

// POST /orders — the handler, with every failure mode from this page in it.
func (a *App) PlaceOrder(w http.ResponseWriter, r *http.Request) {
    ctx, cancel := context.WithTimeout(r.Context(), 8*time.Second)
    defer cancel()

    key := r.Header.Get("Idempotency-Key")
    if key == "" {
        writeErr(w, 400, "idempotency_key_required")
        return
    }

    var req PlaceOrderRequest
    if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, 1<<20)).Decode(&req); err != nil {
        writeErr(w, 400, "bad_json")
        return
    }

    // 1. Claim the key. Conditional put: whoever wins does the work, and the
    //    TTL means abandoned claims clean themselves up.
    claimed, prior, err := a.claim(ctx, key, 24*time.Hour)
    switch {
    case err != nil:
        writeErr(w, 503, "try_again")
        return
    case !claimed && prior != nil:
        w.Header().Set("Idempotent-Replay", "true")
        _, _ = w.Write(prior)                    // byte-identical to the first reply
        return
    case !claimed:
        writeErr(w, 409, "in_progress")          // claimed, not finished: retry later
        return
    }

    // 2. Order and outbox row commit together, or neither exists.
    tx, err := a.pool.Begin(ctx)
    if err != nil {
        writeErr(w, 503, "try_again")
        return
    }
    defer tx.Rollback(ctx)                        // no-op after a successful commit

    order, err := insertOrder(ctx, tx, req)
    if err != nil {
        writeErr(w, 500, "internal")
        return
    }
    event, _ := json.Marshal(OrderPlaced{Version: "1", OrderID: order.ID})
    if err := insertOutbox(ctx, tx, order.ID, "OrderPlaced", event); err != nil {
        writeErr(w, 500, "internal")
        return
    }
    if err := tx.Commit(ctx); err != nil {
        // A commit that fails after a failover may still have applied. The
        // idempotency key is what makes the client’s retry safe here.
        writeErr(w, 503, "try_again")
        return
    }

    // 3. Store the reply against the key, so the retry gets THIS body.
    body, _ := json.Marshal(order)
    _ = a.completeClaim(ctx, key, body)
    w.WriteHeader(201)
    _, _ = w.Write(body)
}

@app.post("/orders", status_code=201)
async def place_order(req: PlaceOrderRequest, response: Response,
                      idempotency_key: str = Header(alias="Idempotency-Key")):

    # 1. Claim the key. Conditional put: whoever wins does the work, and the
    #    TTL means abandoned claims clean themselves up.
    claimed, prior = await claim(idempotency_key, ttl=timedelta(hours=24))
    if not claimed and prior is not None:
        response.headers["Idempotent-Replay"] = "true"
        return Response(content=prior, media_type="application/json")
    if not claimed:
        raise HTTPException(409, "in_progress")   # claimed, not finished

    # 2. Order and outbox row commit together, or neither exists.
    try:
        async with engine.begin() as conn:        # commits on exit, rolls back on raise
            order = await insert_order(conn, req)
            await conn.execute(
                text("""INSERT INTO outbox (aggregate_id, type, payload)
                        VALUES (:id, :type, CAST(:payload AS jsonb))"""),
                {"id": order.id, "type": "OrderPlaced",
                 "payload": json.dumps({"version": "1", "orderId": order.id})},
            )
    except OperationalError:
        # A commit that fails after a failover may still have applied. The
        # idempotency key is what makes the client’s retry safe here.
        raise HTTPException(503, "try_again")

    # 3. Store the reply against the key, so the retry gets THIS body.
    body = order.model_dump_json()
    await complete_claim(idempotency_key, body)
    return Response(content=body, status_code=201, media_type="application/json")

The claim, in DynamoDB

One conditional write does all of it: reserve the key if nobody has it, return the stored response if the work is finished, and allow a re-attempt if a previous holder died before completing.

// claim reserves the idempotency key, or hands back the finished response.
func (a *App) claim(ctx context.Context, key string, ttl time.Duration) (bool, []byte, error) {
    now := time.Now()
    _, err := a.ddb.PutItem(ctx, &dynamodb.PutItemInput{
        TableName: aws.String("idempotency"),
        Item: map[string]types.AttributeValue{
            "pk":       &types.AttributeValueMemberS{Value: key},
            "state":    &types.AttributeValueMemberS{Value: "in_progress"},
            "expires":  &types.AttributeValueMemberN{Value: fmt.Sprint(now.Add(ttl).Unix())},
            "lockedAt": &types.AttributeValueMemberN{Value: fmt.Sprint(now.Unix())},
        },
        // Win if the key is unknown, OR if a previous holder crashed mid-flight
        // and its lock is stale. A FINISHED claim is never retaken — without the
        // state check, a retry 30s later re-runs the order instead of replaying it.
        // ("state" is a DynamoDB reserved word, hence the #st alias.)
        ConditionExpression: aws.String(
            "attribute_not_exists(pk) OR (#st = :inprog AND lockedAt < :stale)"),
        ExpressionAttributeNames: map[string]string{"#st": "state"},
        ExpressionAttributeValues: map[string]types.AttributeValue{
            ":inprog": &types.AttributeValueMemberS{Value: "in_progress"},
            ":stale": &types.AttributeValueMemberN{
                Value: fmt.Sprint(now.Add(-30 * time.Second).Unix())},
        },
        ReturnValuesOnConditionCheckFailure: types.ReturnValuesOnConditionCheckFailureAllOld,
    })
    if err == nil {
        return true, nil, nil
    }

    var failed *types.ConditionalCheckFailedException
    if !errors.As(err, &failed) {
        return false, nil, err
    }
    // We lost. The existing item comes back with the failure, so a completed
    // request needs no second read.
    if resp, ok := failed.Item["response"].(*types.AttributeValueMemberB); ok {
        return false, resp.Value, nil
    }
    return false, nil, nil                 // still in progress elsewhere
}
# claim() reserves the idempotency key, or hands back the finished response.
async def claim(key: str, ttl: timedelta) -> tuple[bool, bytes | None]:
    now = int(time.time())
    try:
        DDB.put_item(
            TableName="idempotency",
            Item={
                "pk": {"S": key},
                "state": {"S": "in_progress"},
                "expires": {"N": str(now + int(ttl.total_seconds()))},
                "lockedAt": {"N": str(now)},
            },
            # Win if the key is unknown, OR if a previous holder crashed mid-flight
            # and its lock is stale. A FINISHED claim is never retaken — without the
            # state check, a retry 30s later re-runs the order instead of replaying it.
            # ("state" is a DynamoDB reserved word, hence the #st alias.)
            ConditionExpression=(
                "attribute_not_exists(pk) OR (#st = :inprog AND lockedAt < :stale)"
            ),
            ExpressionAttributeNames={"#st": "state"},
            ExpressionAttributeValues={":inprog": {"S": "in_progress"},
                                       ":stale": {"N": str(now - 30)}},
            ReturnValuesOnConditionCheckFailure="ALL_OLD",
        )
        return True, None
    except DDB.exceptions.ConditionalCheckFailedException as e:
        # We lost. The existing item comes back with the failure, so a completed
        # request needs no second read.
        item = e.response.get("Item") or {}
        stored = item.get("response", {}).get("B")
        return False, stored          # stored is None while still in progress

The checklist, and one script that proves it

Everything above, compressed into questions you can ask at a design review — and a start-up self-test that turns half of them from a promise into an assertion.

// preflight runs before the readiness probe goes green. Everything it checks
// is something that has silently broken a production deploy at least once.
func preflight(ctx context.Context, a *App) error {
    ctx, cancel := context.WithTimeout(ctx, 10*time.Second)
    defer cancel()

    // 1. Am I who the deployment thinks I am?
    who, err := a.sts.GetCallerIdentity(ctx, nil)
    if err != nil {
        return fmt.Errorf("no usable credentials: %w", err)
    }
    if !strings.Contains(aws.ToString(who.Arn), expectedRole) {
        return fmt.Errorf("wrong role: %s", aws.ToString(who.Arn))
    }

    // 2. Does the table exist, in this region, with a TTL as designed?
    tt, err := a.ddb.DescribeTimeToLive(ctx, &dynamodb.DescribeTimeToLiveInput{
        TableName: aws.String("idempotency"),
    })
    if err != nil {
        return fmt.Errorf("idempotency table: %w", err)
    }
    if tt.TimeToLiveDescription.TimeToLiveStatus != types.TimeToLiveStatusEnabled {
        return errors.New("idempotency table has no TTL: it will grow for ever")
    }

    // 3. Does every queue we consume have somewhere for poison to go?
    for _, q := range a.queues {
        at, err := a.sqs.GetQueueAttributes(ctx, &sqs.GetQueueAttributesInput{
            QueueUrl:       aws.String(q),
            AttributeNames: []sqstypes.QueueAttributeName{"RedrivePolicy"},
        })
        if err != nil {
            return err
        }
        if at.Attributes["RedrivePolicy"] == "" {
            return fmt.Errorf("queue %s has no dead letter queue", q)
        }
    }

    // 4. Can we actually write, with the timeout we promised?
    return a.pool.QueryRow(ctx, "SELECT 1").Scan(new(int))
}

The four sentences worth remembering

Every call can be retried, so every effect must be idempotent. Every managed service has a limit that is per-partition, per-shard or per-connection, not per-account — find yours before it finds you. The default configuration is optimised for getting started, not for running. And the bill is mostly data movement: bytes crossing a boundary you did not think about cost more than the compute that produced them.
