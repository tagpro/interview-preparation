Part seven

Running it

Even if you never serve a model yourself — and most AI engineers never will — this part decides what your API bill looks like, why latency behaves the way it does, and what the person on the other side of the interview means when they ask you to design a batching system. Four topics: what has to fit in memory, how a GPU is shared between users, what quantisation buys, and the two latencies a user actually feels.

AI · xx

What has to fit on the card

Three things share GPU memory: the weights, the KV cache, and everything transient. Only the first is a constant. The second scales with context length times concurrency, and it is the number that decides how many users you can serve at once — which is why the shape of the attention is a serving decision, not a research one.

The problemA 16 GB model on an 80 GB card falls over at thirty concurrent users, and nobody can say why, because the weights obviously fit.
The mechanismKV cache bytes = 2 × layers × kv-heads × head-dim × 2 bytes, per token, per sequence. Multiply by context length and concurrency and it dwarfs the weights.
In the room“Weights are fixed, so the interesting budget is what is left over divided by the per-token cache cost. That division is your maximum concurrency, and it is the number I would size a fleet on.”
ConsumerSizeScales with
Weightsparameters × bytes-per-weightNothing. Roughly 2 GB per billion parameters at bf16, 0.5 at 4-bit.
KV cache2 × L × kv-heads × head-dim × 2 bytes per tokenContext length × concurrency. The one that surprises people.
ActivationsBatch × sequence × hiddenBatch size during prefill. Modest during decode.
Fragmentation and runtimeA few percent, plus graphs and buffersPaged allocators exist precisely to reclaim this.
L1 · Beginner

The back of the envelope

  • ~2 GB per billion parameters at bf16, half that at fp8, a quarter at 4-bit.
  • The KV cache is per user, per token, and it does not go away until the sequence ends.
  • Leave headroom. A card at 98% is a card that will out-of-memory on a long prompt.
  • Long context and high concurrency are the same budget. You choose one.
L2 · Intermediate

Make the cache cheaper

  • Grouped-query attention is the biggest single lever, and it is chosen when the model is trained, not when it is served.
  • Paged allocation stores the cache in fixed blocks, which removes the fragmentation that a contiguous allocation forces.
  • Prefix sharing lets many requests with the same system prompt share one copy of its cache.
  • Quantising the cache itself to fp8 halves it again, at a small quality cost.
L3 · Advanced

When one card is not enough

  • Tensor parallelism splits each layer across cards: lower latency, and it needs fast interconnect.
  • Pipeline parallelism splits by layer: better throughput, higher latency, tolerant of slower links.
  • Prefill and decode can be disaggregated onto different machines, since one is compute-bound and the other memory-bound.
  • A MoE model breaks the usual arithmetic: all the parameters must be resident in VRAM, but only a fraction of them run per token, so memory scales with the total and compute with the active subset.
  • Offloading the cache to host memory buys capacity and costs latency on every token that needs it.

# capacity.py — the sizing calculation, in the form you would actually put in
# a capacity document. No framework, no benchmark: just the formula and the
# published shape of the model.
from dataclasses import dataclass

GB = 1024 ** 3


@dataclass(frozen=True)
class Shape:
    name: str
    params: float
    layers: int
    kv_heads: int      # 8 on a grouped-query model, 32+ on multi-head
    head_dim: int = 128


LLAMA_8B = Shape("Llama 3.1 8B", 8.03e9, 32, 8)
LLAMA_70B = Shape("Llama 3.1 70B", 70.6e9, 80, 8)


def kv_bytes_per_token(s: Shape, cache_bits: int = 16) -> int:
    """2 for K and V, then layers x kv_heads x head_dim, at the cache's precision."""
    return 2 * s.layers * s.kv_heads * s.head_dim * (cache_bits // 8)


def max_concurrency(s: Shape, card_gb: float, context: int,
                    weight_bits: int = 16, cache_bits: int = 16) -> int:
    weights = s.params * weight_bits / 8
    overhead = weights * 0.06 + 2 * GB          # runtime, graphs, fragmentation
    free = card_gb * GB - weights - overhead
    if free <= 0:
        return 0
    return int(free // (kv_bytes_per_token(s, cache_bits) * context))

// admission.ts — the control that turns "out of memory" into "please wait".
// A server that accepts a request it cannot hold will kill an in-flight one to
// make room, which turns one user's long prompt into everyone's failure.

export interface Budget {
  totalCacheBytes: number;   // free memory after weights and overhead
  bytesPerToken: number;     // from the model's shape
}

export class Admission {
  private held = new Map<string, number>();
  private used = 0;

  constructor(private budget: Budget, private reserve = 0.1) {}

  /** Reserve worst-case memory up front: prompt plus the maximum output.
   *  Reserving only the prompt is how a server dies two thousand tokens into
   *  a generation it had already accepted. */
  admit(id: string, promptTokens: number, maxOutput: number): boolean {
    const need = (promptTokens + maxOutput) * this.budget.bytesPerToken;
    const ceiling = this.budget.totalCacheBytes * (1 - this.reserve);
    if (this.used + need > ceiling) return false;
    this.held.set(id, need);
    this.used += need;
    return true;
  }

  release(id: string): void {
    this.used -= this.held.get(id) ?? 0;
    this.held.delete(id);
  }

  get utilisation(): number {
    return this.used / this.budget.totalCacheBytes;
  }
}

// Being able to do this on a whiteboard is a genuinely common interview ask,
// and the formula is short enough to memorise.
package capacity

const gb = 1 << 30

type Shape struct {
	Name    string
	Params  float64
	Layers  int
	KVHeads int
	HeadDim int
}

// KVBytesPerToken: 2 (K and V) x layers x kv_heads x head_dim x 2 bytes (fp16).
func (s Shape) KVBytesPerToken() int {
	return 2 * s.Layers * s.KVHeads * s.HeadDim * 2
}

// WeightBytes at the given bits per weight: 16 for bf16, 8 for fp8, 4 for int4.
func (s Shape) WeightBytes(bits int) float64 {
	return s.Params * float64(bits) / 8
}

// Concurrency is the whole question, in one line: what is left after the
// weights, divided by what one sequence costs.
func (s Shape) Concurrency(cardBytes float64, bits, context int) int {
	weights := s.WeightBytes(bits)
	free := cardBytes - weights - weights*0.06 - 2*gb
	if free <= 0 {
		return 0
	}
	return int(free / float64(s.KVBytesPerToken()*context))
}

Sharing a GPU

Generating one token for one user leaves a GPU almost entirely idle: it is a memory-bound operation on a device built for arithmetic. The fix is to generate a token for many sequences at once. The subtlety is when a finished sequence releases its slot, and that single scheduling decision is most of the difference between a naive server and a real one.

The problemA static batch waits for its slowest member. Seven requests that needed four tokens sit finished in their slots while the eighth generates its four-hundredth, and the queue behind them waits.
The mechanismContinuous batching makes the scheduling unit a single decode step. A slot frees the instant a sequence emits a stop token and is refilled on the next step.
In the room“Iteration-level scheduling, paged KV cache so a freed slot is genuinely reusable, and chunked prefill so one long prompt does not stall everyone else’s decoding.”
TechniqueWhat it fixesCost
Continuous batchingHead-of-line blocking from finished sequencesNone worth mentioning. It is the baseline expectation now.
Paged KV cacheFragmentation, so freed memory is actually reusableA small indirection per attention step; it pays for itself many times over.
Prefix sharingRe-processing an identical system prompt for every requestBookkeeping. Huge win when one prompt is shared across all traffic.
Chunked prefillA long prompt blocking everyone else’s decode stepsSlightly lower prefill throughput, much better tail latency.
Priority queuesAn interactive request stuck behind a batch jobComplexity, and the risk of starving the low-priority queue.

Why throughput and latency pull in opposite directions

A larger batch means more sequences share each forward pass, so tokens per second across the fleet goes up. It also means each individual sequence waits behind more work, so tokens per second for one user goes down. There is no setting that maximises both, and the right answer depends on which you are selling: an interactive assistant optimises for the second, a nightly document pipeline for the first. Say which one you are optimising before you quote a number, because “it does 3,000 tokens a second” is meaningless without it.

Snippets in Python, TypeScript and Go.

# scheduler.py — the difference between the two schedulers, in about forty
# lines, with the scheduling unit as the only thing that changes.
from dataclasses import dataclass, field


@dataclass
class Request:
    id: int
    arrives: int
    output_len: int
    started: int = -1
    finished: int = -1


def simulate(requests: list[Request], slots: int, continuous: bool) -> dict:
    pending = sorted(requests, key=lambda r: r.arrives)
    running: list[Request | None] = [None] * slots
    remaining = {r.id: r.output_len for r in requests}
    busy = idle = step = 0

    while any(r.finished < 0 for r in requests):
        # THE line. A static batch may only admit work when every slot is free;
        # a continuous one refills any slot the moment it opens.
        may_admit = continuous or all(s is None for s in running)
        if may_admit:
            for i in range(slots):
                if running[i] is not None:
                    continue
                nxt = next((r for r in pending if r.arrives <= step and r.started < 0), None)
                if nxt is None:
                    break
                nxt.started = step
                running[i] = nxt

        for i, r in enumerate(running):
            if r is None or r.finished >= 0:
                idle += 1                     # a slot doing no useful work
                continue
            busy += 1
            remaining[r.id] -= 1
            if remaining[r.id] <= 0:
                r.finished = step
                if continuous:
                    running[i] = None         # released immediately

        if not continuous and all(r is None or r.finished >= 0 for r in running):
            running = [None] * slots          # the whole batch drains together

        step += 1
        if step > 10_000:
            break

    return {"steps": step, "utilisation": busy / (busy + idle)}


reqs = [Request(i, a, n) for i, (a, n) in
        enumerate([(0, 6), (0, 22), (0, 4), (1, 9), (3, 5), (5, 14), (8, 3), (9, 7)])]
print("static    ", simulate([Request(**vars(r)) for r in reqs], 4, continuous=False))
print("continuous", simulate([Request(**vars(r)) for r in reqs], 4, continuous=True))

# static     {'steps': 46, 'utilisation': 0.38}
# continuous {'steps': 22, 'utilisation': 0.80}

// client.ts — most engineers meet batching from the client side, where the
// question is how many requests to have in flight. Too few wastes the
// provider's throughput; too many earns rate limits and retry storms.

export interface Limits {
  concurrency: number;       // requests in flight
  tokensPerMinute: number;   // the constraint that actually binds
}

export class Pool<T> {
  private inFlight = 0;
  private windowStart = Date.now();
  private tokensThisWindow = 0;
  private queue: (() => void)[] = [];

  constructor(private limits: Limits) {}

  async run(estimatedTokens: number, task: () => Promise<T>): Promise<T> {
    await this.acquire(estimatedTokens);
    try {
      return await task();
    } finally {
      this.inFlight--;
      this.queue.shift()?.();
    }
  }

  private async acquire(tokens: number): Promise<void> {
    // Reset the token window every minute. Tracking tokens rather than just
    // request count is the part people skip, and token limits are usually what
    // you hit first on a workload with large prompts.
    if (Date.now() - this.windowStart > 60_000) {
      this.windowStart = Date.now();
      this.tokensThisWindow = 0;
    }
    while (
      this.inFlight >= this.limits.concurrency ||
      this.tokensThisWindow + tokens > this.limits.tokensPerMinute
    ) {
      await new Promise<void>((resolve) => this.queue.push(resolve));
      if (Date.now() - this.windowStart > 60_000) {
        this.windowStart = Date.now();
        this.tokensThisWindow = 0;
      }
    }
    this.inFlight++;
    this.tokensThisWindow += tokens;
  }
}

// retry.go — the other half of talking to a rate-limited service: backing off
// correctly. Honour the server's own Retry-After header, add jitter, and cap
// total wall-clock rather than attempt count, because a caller waiting three
// minutes does not care that it was "only four attempts".
package llm

import (
	"context"
	"errors"
	"math"
	"math/rand"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

type Backoff struct {
	Base     time.Duration // 500ms is a reasonable start
	Max      time.Duration // per-attempt ceiling
	Deadline time.Duration // total wall-clock across all attempts
}

func Do[T any](ctx context.Context, b Backoff, call func(context.Context) (T, error)) (T, error) {
	var zero T
	ctx, cancel := context.WithTimeout(ctx, b.Deadline)
	defer cancel()

	for attempt := 0; ; attempt++ {
		out, err := call(ctx)
		if err == nil {
			return out, nil
		}

		var apiErr *anthropic.Error
		if !errors.As(err, &apiErr) || !retryable(apiErr.StatusCode) {
			return zero, err // a 400 will still be a 400 in two seconds
		}

		// Full jitter: without it, every client that was rate-limited together
		// retries together, and the thundering herd re-creates the overload.
		wait := time.Duration(float64(b.Base) * math.Pow(2, float64(attempt)))
		if wait > b.Max {
			wait = b.Max
		}
		wait = time.Duration(rand.Int63n(int64(wait)))

		select {
		case <-ctx.Done():
			return zero, ctx.Err()
		case <-time.After(wait):
		}
	}
}

func retryable(status int) bool {
	return status == 408 || status == 409 || status == 429 || status >= 500
}

Fewer bits

A weight stored in sixteen bits can usually be stored in eight, and often in four, with a quality loss that is hard to measure. Since serving is memory-bound, fewer bits means less to move, which means faster as well as smaller. The interesting question is not whether to quantise but where the loss shows up, because it is not spread evenly across tasks.

The problemThe model needs two cards. Two cards cost twice as much, add an interconnect hop to every token, and are harder to schedule.
The mechanismStore weights at lower precision with a scale factor per group, and dequantise on the fly in the kernel. The arithmetic still happens at higher precision; only the storage shrinks.
In the room“fp8 where the hardware supports it natively, 4-bit when I need the model to fit at all. And I would measure on my task, because benchmark perplexity hides the degradation on long-form reasoning.”
FormatMemoryQuality and caveats
bf16 / fp162 bytes per weightThe reference. Roughly 2 GB per billion parameters.
fp81 byteNative on recent data-centre GPUs. Keeps floating-point dynamic range; usually the best quality per byte when supported.
int81 byteWell understood, widely supported, typically imperceptible with sensible calibration.
int4 (AWQ, GPTQ, and similar)0.5 bytesThree-quarters saved. Measurable degradation on reasoning-heavy work; excellent for fitting a bigger model on one card.
KV cache quantisationHalves the cacheIndependent of weight quantisation and often the more useful one, since the cache is what limits concurrency.

Perplexity is the wrong test

Quantisation barely moves perplexity on general text long after it has started hurting on the tasks people care about — multi-step reasoning, code, long-context recall, and instruction-following at the edges. It also degrades non-English output before English. If you are going to quantise, evaluate on your own task suite before and after, and pay particular attention to the hardest ten percent of your cases. “Perplexity moved by 1%” is a sentence that has shipped a lot of quietly worse systems.

# quantise.py — group-wise int8, which is the whole idea: one scale factor per
# small group of weights rather than one for the whole tensor, so a single
# outlier cannot flatten everything around it.
import torch


def quantise(w: torch.Tensor, group: int = 128) -> tuple[torch.Tensor, torch.Tensor]:
    """Symmetric int8 with a per-group scale. Returns codes and scales."""
    out_features, in_features = w.shape
    assert in_features % group == 0

    grouped = w.reshape(out_features, in_features // group, group)
    scale = grouped.abs().amax(dim=-1, keepdim=True) / 127.0
    scale = scale.clamp(min=1e-8)
    codes = (grouped / scale).round().clamp(-127, 127).to(torch.int8)
    return codes.reshape(out_features, in_features), scale.squeeze(-1)


def dequantise(codes: torch.Tensor, scale: torch.Tensor, group: int = 128) -> torch.Tensor:
    out_features, in_features = codes.shape
    grouped = codes.reshape(out_features, in_features // group, group).float()
    return (grouped * scale.unsqueeze(-1)).reshape(out_features, in_features)


def error(w: torch.Tensor, group: int) -> float:
    codes, scale = quantise(w, group)
    return (dequantise(codes, scale, group) - w).abs().mean().item()


w = torch.randn(4096, 4096)
w[0, 0] = 40.0                          # one outlier, of the kind real weights have
for g in (4096, 512, 128, 32):
    print(f"group={g:>5}  mean abs error {error(w, g):.5f}")

// tradeoff.ts — the sizing decision as a table you can put in a document,
// because "will it fit" and "how much worse" are the only two questions.

export interface Option {
  label: string;
  bitsPerWeight: number;
  expectedQualityLoss: string; // qualitative on purpose: measure your own
}

export const OPTIONS: Option[] = [
  { label: "bf16", bitsPerWeight: 16, expectedQualityLoss: "reference" },
  { label: "fp8", bitsPerWeight: 8, expectedQualityLoss: "negligible on supported hardware" },
  { label: "int8", bitsPerWeight: 8, expectedQualityLoss: "negligible with calibration" },
  { label: "int4", bitsPerWeight: 4, expectedQualityLoss: "measurable on reasoning tasks" },
];

export function fits(paramsBillions: number, cardGB: number, o: Option, overhead = 1.15): boolean {
  return (paramsBillions * o.bitsPerWeight) / 8 * overhead < cardGB;
}

export function plan(paramsBillions: number, cardGB: number) {
  return OPTIONS.map((o) => ({
    ...o,
    weightsGB: +((paramsBillions * o.bitsPerWeight) / 8).toFixed(1),
    fits: fits(paramsBillions, cardGB, o),
  }));
}

// gate.go — quantisation is a change to the model, so it goes through the
// same gate as any other: measure on your own suite, per category, and refuse
// to promote on an aggregate that hides a regression.
package quant

import "fmt"

type CategoryScore struct {
	Category string
	Baseline float64
	Candidate float64
	N        int
}

type Decision struct {
	Promote bool
	Reasons []string
}

// Evaluate refuses on any single category that drops more than the tolerance,
// even when the average improves. Quantisation damage is concentrated — long
// reasoning, code, non-English — and an average across categories is exactly
// the statistic that hides it.
func Evaluate(scores []CategoryScore, tolerance float64) Decision {
	var d Decision
	d.Promote = true

	var sumBase, sumCand float64
	for _, s := range scores {
		sumBase += s.Baseline
		sumCand += s.Candidate
		if drop := s.Baseline - s.Candidate; drop > tolerance {
			d.Promote = false
			d.Reasons = append(d.Reasons, fmt.Sprintf(
				"%s fell %.1f points (%.1f -> %.1f, n=%d)",
				s.Category, drop*100, s.Baseline*100, s.Candidate*100, s.N))
		}
	}
	d.Reasons = append(d.Reasons, fmt.Sprintf("aggregate %.1f -> %.1f",
		sumBase/float64(len(scores))*100, sumCand/float64(len(scores))*100))
	return d
}

The two latencies a user feels

There are two numbers, not one. Time to first token is how long the screen stays empty; it is dominated by prefill, which is quadratic in prompt length. Time per output token is how fast the answer then appears; it is dominated by decode, which is memory-bound and roughly constant. Users perceive these completely differently, and optimising them requires different work.

The problem“It feels slow”, reported against a system whose total latency is fine. Seven seconds of blank screen followed by an instant answer feels worse than one second followed by six of steady text.
The mechanismStreaming decouples the two. A user reads at fifteen to twenty tokens a second, so anything above that is invisible — but every millisecond before the first token is felt in full.
In the room“I would report TTFT and tokens per second separately, at p50 and p99, and I would stream. Total latency is an average that hides the number the user actually experiences.”
LeverWhich latencyEffect
Stream the responsePerceivedThe largest single win, and usually the cheapest. Do this first.
Shorten the promptFirst tokenPrefill is superlinear in prompt length. Halving retrieval halves the empty screen.
Cache the prefixFirst tokenA cached prefix skips most of prefill, not just the billing for it.
Ask for a shorter answerTotalOutput tokens are generated one at a time. This is the only lever on the second half.
Speculative decodingPer tokenA small draft model proposes several tokens; the big one verifies them in a single pass. Output is identical; throughput can double.
Route to a smaller modelBothFewer layers, less memory traffic, faster on both halves.
Do the work in parallelWall clockRetrieval and a safety check can run concurrently with each other, not in series before the model call.

Speculative decoding, in one paragraph

Decoding is memory-bound: the GPU spends its time moving weights, not multiplying. Verifying five candidate tokens costs almost the same as generating one, because the weights are moved once either way. So a small fast draft model proposes the next few tokens, the large model checks them all in a single forward pass, and every accepted token was free. Rejections cost one wasted pass and the sequence continues from the divergence, so the output is exactly what the large model would have produced — this is a pure latency optimisation with no quality trade. Acceptance rate is everything: a well-matched draft model gets sixty to eighty percent, and a badly matched one makes things slower.

Snippets in Python, TypeScript and Go.

# stream.py — streaming, and the measurement that goes with it. Recording time
# to first token separately is the whole point: an average latency number
# cannot distinguish a fast start from a fast finish, and users can.
import time
from anthropic import Anthropic

client = Anthropic()


def ask(prompt: str) -> dict:
    started = time.perf_counter()
    first = None
    chunks = []

    with client.messages.stream(
        model="claude-opus-5",
        max_tokens=4096,
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        for text in stream.text_stream:
            if first is None:
                first = time.perf_counter()          # the number users feel
            chunks.append(text)
            print(text, end="", flush=True)
        final = stream.get_final_message()

    done = time.perf_counter()
    out_tokens = final.usage.output_tokens
    return {
        "ttft_ms": round((first - started) * 1000) if first else None,
        "total_ms": round((done - started) * 1000),
        # Tokens per second AFTER the first one: the generation rate, undiluted
        # by however long prefill took.
        "tokens_per_sec": round(out_tokens / max(1e-6, done - first), 1) if first else None,
        "output_tokens": out_tokens,
    }

# {'ttft_ms': 780, 'total_ms': 7240, 'tokens_per_sec': 46.3, 'output_tokens': 299}

// parallel.ts — the latency win that needs no model change: stop doing in
// series what can be done at once. Retrieval, a safety pre-check and any
// enrichment are independent, so the wall-clock cost should be the slowest of
// them, not their sum.
import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

export interface Deps {
  retrieve(q: string): Promise<string[]>;
  screenInput(q: string): Promise<{ blocked: boolean; reason?: string }>;
  userContext(userId: string): Promise<string>;
}

export async function respond(deps: Deps, userId: string, question: string) {
  // 380ms + 120ms + 60ms in series; ~380ms concurrently. The user notices.
  const [passages, screen, profile] = await Promise.all([
    deps.retrieve(question),
    deps.screenInput(question),
    deps.userContext(userId),
  ]);

  if (screen.blocked) return { blocked: true, reason: screen.reason };

  const stream = client.messages.stream({
    model: "claude-opus-5",
    max_tokens: 4096,
    system: [{ type: "text", text: profile, cache_control: { type: "ephemeral" } }],
    messages: [
      { role: "user", content: `${passages.join("\n\n")}\n\nQuestion: ${question}` },
    ],
  });

  // Return the stream rather than awaiting it. Buffering a complete answer
  // before sending it to the browser throws away every millisecond streaming
  // just bought you — a mistake that is easy to make one layer at a time.
  return { blocked: false, stream };
}

}

type Budget struct {
	TimeToFirstToken time.Duration // what the user actually feels
	Stages           []Stage       // everything before the model call
}

func (b Budget) Validate() error {
	var sum time.Duration
	for _, s := range b.Stages {
		sum += s.Budget
	}
	if sum >= b.TimeToFirstToken {
		return fmt.Errorf("stages total %v against a %v first-token budget: no room for prefill",
			sum, b.TimeToFirstToken)
	}
	return nil
}

// Run gives each stage its own deadline so one slow dependency degrades that
// stage rather than the whole request. Retrieval that times out should produce
// an answer without sources, not no answer at all.
func (b Budget) Run(ctx context.Context, s Stage, f func(context.Context) error) error {
	sctx, cancel := context.WithTimeout(ctx, s.Budget)
	defer cancel()
	if err := f(sctx); err != nil {
		return fmt.Errorf("%s (budget %v): %w", s.Name, s.Budget, err)
	}
	return nil
}
