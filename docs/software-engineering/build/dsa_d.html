Part four

Sorting, searching and selecting

You will not write a sort. You will, constantly, decide whether to sort — and that decision needs you to know what sorting costs, what it buys, what your language actually runs when you call it, and which of the three cheaper questions (is it here? where would it go? what are the top ten?) you could have asked instead.

Six sorts, and what separates them

Every comparison sort does the same job with the same lower bound, so what distinguishes them is not the exponent. It is the memory they need, whether equal elements keep their original order, how they behave on data that is already nearly sorted, and how they fail on the input designed to hurt them.

Stability is a correctness property, not a nicety

A stable sort keeps equal elements in their original relative order. That is what lets you sort by one field, then by another, and end up with a result ordered by both — sort by name, then by department, and each department’s names are still alphabetical. With an unstable sort the second pass scrambles the first, and the bug shows up as “the list is mostly right”, which is the hardest kind to notice. Python’s sort is stable, Java’s Collections.sort and Arrays.sort on objects are stable, Go’s sort.Slice and slices.Sort are not — Go makes you ask for slices.SortStableFunc.

// mergesort.go — the one worth being able to write from memory, because
// the recurrence T(n) = 2T(n/2) + O(n) is the shape of half this page.

func MergeSort(xs []int) []int {
    if len(xs) < 2 {
        return xs
    }
    mid := len(xs) / 2
    left := MergeSort(slices.Clone(xs[:mid]))
    right := MergeSort(slices.Clone(xs[mid:]))
    return merge(left, right)
}

// The merge is the algorithm. Note `<=`: taking from the left on a tie
// is exactly what makes the sort stable. `<` would not.
func merge(a, b []int) []int {
    out := make([]int, 0, len(a)+len(b))
    i, j := 0, 0
    for i < len(a) && j < len(b) {
        if a[i] <= b[j] {
            out, i = append(out, a[i]), i+1
        } else {
            out, j = append(out, b[j]), j+1
        }
    }
    return append(append(out, a[i:]...), b[j:]...)
}

// Quicksort's partition, for contrast: no buffer, and the whole cost is
// in choosing the pivot. Median-of-three on the first, middle and last
// element turns "sorted input is O(n^2)" into "you would have to try".
func partition(xs []int, lo, hi int) int {
    mid := lo + (hi-lo)/2
    if xs[mid] < xs[lo] { xs[mid], xs[lo] = xs[lo], xs[mid] }
    if xs[hi] < xs[lo]  { xs[hi], xs[lo] = xs[lo], xs[hi] }
    if xs[hi] < xs[mid] { xs[hi], xs[mid] = xs[mid], xs[hi] }
    pivot := xs[mid]
    i, j := lo, hi
    for i <= j {
        for xs[i] < pivot { i++ }
        for xs[j] > pivot { j-- }
        if i <= j {
            xs[i], xs[j] = xs[j], xs[i]
            i, j = i+1, j-1
        }
    }
    return i
}
# sorts.py — write these once to understand them, then never again.

def merge_sort(xs: list[int]) -> list[int]:
    if len(xs) < 2:
        return xs
    mid = len(xs) // 2
    left, right = merge_sort(xs[:mid]), merge_sort(xs[mid:])

    out, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:                  # `<=` is what makes it stable
            out.append(left[i]); i += 1
        else:
            out.append(right[j]); j += 1
    out.extend(left[i:]); out.extend(right[j:])
    return out

# Three-way (Dutch national flag) partitioning: the fix for quicksort on
# data with many duplicates. Everything equal to the pivot ends up in
# the middle and is never looked at again, so an array of one repeated
# value goes from O(n^2) to O(n).
def quicksort(xs: list[int]) -> list[int]:
    if len(xs) < 2:
        return xs
    pivot = xs[len(xs) // 2]
    lt = [x for x in xs if x < pivot]
    eq = [x for x in xs if x == pivot]
    gt = [x for x in xs if x > pivot]
    return quicksort(lt) + eq + quicksort(gt)

# Counting sort: no comparisons, so the n log n bound simply does not
# apply. O(n + k) for keys in a small known range -- ages, HTTP status
# codes, bucket indices, anything with a few thousand distinct values.
def counting_sort(xs: list[int], k: int) -> list[int]:
    counts = [0] * k
    for x in xs:
        counts[x] += 1
    return [v for v, c in enumerate(counts) for _ in range(c)]
// Sorts.java — and the reason the JDK ships two completely different
// algorithms, chosen by the element type.

static void mergeSort(int[] a, int[] buf, int lo, int hi) {
    if (hi - lo < 2) return;
    int mid = (lo + hi) >>> 1;                   // >>> not /2: no overflow
    mergeSort(a, buf, lo, mid);
    mergeSort(a, buf, mid, hi);
    if (a[mid - 1] <= a[mid]) return;            // already ordered: skip the merge

    int i = lo, j = mid, k = lo;
    while (i < mid && j < hi) buf[k++] = (a[i] <= a[j]) ? a[i++] : a[j++];
    while (i < mid) buf[k++] = a[i++];
    while (j < hi) buf[k++] = a[j++];
    System.arraycopy(buf, lo, a, lo, hi - lo);
}

What your language actually runs

None of the six sorts above is what happens when you call sort. All three of these languages ship a hybrid: a main algorithm, a small-array fallback, a detector for pre-existing order, and in two of the three an escape hatch for adversarial input. Knowing which hybrid you have tells you what your worst case really is.

// Multi-key: compare the first, and only fall through on a tie. cmp.Or
// (Go 1.22) returns the first non-zero, which reads exactly like SQL's
// ORDER BY dept, name DESC.
slices.SortFunc(rows, func(a, b Row) int {
    return cmp.Or(
        cmp.Compare(a.Dept, b.Dept),
        cmp.Compare(b.Name, a.Name),             // reversed: b before a
    )
})

    ks[i] = keyed{expensiveKey(it), it}          // exactly n calls

Binary search, and the two bounds you actually need

Everyone can describe binary search and most people write it wrong, because the interesting version is not “find x”. It is “where does x belong” — the first position at or after which the answer lies. That version has no special case for “not found”, it handles duplicates, and it is what every standard library actually gives you.

// bsearch.go — the standard library covers the ordinary cases; the one
// worth understanding is the predicate form.

i, found := slices.BinarySearch(xs, target)      // i is the insertion point
if !found {
    xs = slices.Insert(xs, i, target)            // keeps xs sorted
}

// sort.Search is binary search over a PREDICATE, not over values: it
// returns the smallest i in [0, n) for which f(i) is true, assuming f
// is false then true. That "assuming" is the whole contract.
i = sort.Search(len(xs), func(i int) bool { return xs[i] >= target })   // lower bound
j := sort.Search(len(xs), func(i int) bool { return xs[i] > target })   // upper bound
count := j - i                                   // how many equal `target`

// The powerful use: the array need not exist. Binary search the ANSWER
// of a monotone question. "What is the smallest capacity that finishes
// the work in `limit` hours?" -- feasible() is false, false, ..., true.
lo, hi := 1, maxLoad
best := sort.Search(hi-lo, func(k int) bool { return feasible(lo + k) }) + lo

// The overflow bug that lived in the JDK for nine years and in Bentley's
// published version for twenty:
//   mid := (lo + hi) / 2        // overflows once lo+hi exceeds maxint
//   mid := lo + (hi-lo)/2       // does not
# bsearch.py — bisect IS lower_bound and upper_bound, named after what
# they do to the list rather than after what they answer.
import bisect

lo = bisect.bisect_left(xs, target)              # first index with xs[i] >= target
hi = bisect.bisect_right(xs, target)             # first index with xs[i] >  target
count = hi - lo                                  # occurrences, in O(log n)
present = lo < len(xs) and xs[lo] == target      # the "found" test, written out

# Binary search on the answer, written by hand because there is no
# array: the classic "minimum capacity to ship in D days" shape.
def min_feasible(lo: int, hi: int, ok) -> int:
    """Smallest v in [lo, hi] with ok(v) true, where ok is monotone."""
    while lo < hi:
        mid = (lo + hi) // 2                     # Python ints never overflow
        if ok(mid):
            hi = mid                             # mid might be the answer: keep it
        else:
            lo = mid + 1                         # mid is not: discard it
    return lo

# The invariant to hold in your head, and the reason this loop is right:
#   ok(x) is false for every x < lo, and true for hi (or hi is the
#   past-the-end sentinel). Each iteration shrinks the window and
#   preserves that, so when lo == hi it is the answer.
#
# Write `hi = mid` and `lo = mid + 1`. The asymmetry is not a typo: it
# is what guarantees the window shrinks and the loop terminates.
// BSearch.java — Arrays.binarySearch encodes "not found" in the sign,
// which is efficient and catches people out for a decade.

int i = Arrays.binarySearch(xs, target);
if (i < 0) {
    int insertionPoint = -(i + 1);               // the documented decoding
    // negative means absent; -(i+1) is where it would go
}

// It also does NOT promise which one it finds among duplicates. For
// bounds, write the loop -- it is five lines and it is the version you
// want anyway.
static int lowerBound(int[] a, int target) {     // first index with a[i] >= target
    int lo = 0, hi = a.length;                   // note: hi is EXCLUSIVE
    while (lo < hi) {
        int mid = (lo + hi) >>> 1;               // unsigned shift: no overflow
        if (a[mid] < target) lo = mid + 1;
        else hi = mid;
    }
    return lo;                                   // == a.length when all are smaller
}

Selection: the top ten without sorting the other million

“Give me the ten highest scores” does not require the other 999,990 to be put in order. Two algorithms answer it without sorting: a bounded heap, which costs O(n log k) and works on a stream you cannot store, and quickselect, which costs O(n) expected and needs the array in memory.

Sort then sliceO(n log n). Fine up to a few tens of thousands, and the clearest code. Do not optimise past it without a measurement.
Bounded heapO(n log k), O(k) space. The only option when the data is a stream, and the one to use when k is small and n is huge.
QuickselectO(n) expected, in place. Partition, then recurse into one side. The median in linear time, but it reorders your array and its worst case is O(n²).
Partial sortQuickselect then sort k. What you want when the top k must themselves be ordered: O(n + k log k).
// select.go — quickselect is quicksort that throws away half the work.
// After partitioning, the pivot is in its final position; if that is
// past k, only the left side can contain the answer.

func QuickSelect(xs []int, k int) int {          // the k-th smallest, 0-based
    lo, hi := 0, len(xs)-1
    for lo < hi {
        p := partition(xs, lo, hi)               // same partition as quicksort
        switch {
        case p == k:
            return xs[k]
        case p < k:
            lo = p + 1                           // answer is to the right
        default:
            hi = p - 1                           // answer is to the left
        }
    }
    return xs[k]
}

// Why it is O(n): each pass halves the remaining range on average, so
// the total work is n + n/2 + n/4 + ... = 2n. Sorting does log n full
// passes; this does two.
//
// It MUTATES the slice. Clone first if the caller still needs the order.

// The streaming alternative, which never holds more than k:
func TopK(stream iter.Seq[Score], k int) []Score {
    h := &MinHeap{}                              // smallest of the best-k at the root
    for s := range stream {
        if h.Len() < k {
            heap.Push(h, s)
        } else if s.Value > (*h)[0].Value {
            heap.Pop(h)                          // evict the weakest survivor
            heap.Push(h, s)
        }
    }
    out := make([]Score, h.Len())
    for i := len(out) - 1; i >= 0; i-- {
        out[i] = heap.Pop(h).(Score)             // pops ascending, so fill back
    }
    return out
}

import heapq
import random

# nlargest uses a bounded heap when k << n and falls back to sorted()
# when k is a large fraction of n. Trust it.
top = heapq.nlargest(10, scores, key=lambda s: s.value)

# statistics.median is O(n log n) because it sorts. For one median over
# a large array, quickselect is the linear alternative:
def quickselect(xs: list[int], k: int) -> int:
    """k-th smallest, 0-based. Randomised pivot, so sorted input is fine."""
    while True:
        if len(xs) == 1:
            return xs[0]
        pivot = random.choice(xs)                # randomised: no adversarial case
        lt = [x for x in xs if x < pivot]
        eq = [x for x in xs if x == pivot]
        if k < len(lt):
            xs = lt                              # recurse left only
        elif k < len(lt) + len(eq):
            return pivot                         # the pivot IS the answer
        else:
            k -= len(lt) + len(eq)               # recurse right, reindexed
            xs = [x for x in xs if x > pivot]

# For a running median over a stream, neither works: use two heaps, a
# max-heap of the lower half and a min-heap of the upper, kept within
# one element of the same size. The median is then one or two peeks.
// Select.java — the bounded-heap version, which is the one you will
// actually deploy, because it works over an unbounded stream.

static List<Score> topK(Stream<Score> stream, int k) {
    PriorityQueue<Score> heap =                   // MIN-heap of the best k
            new PriorityQueue<>(k, Comparator.comparingInt(Score::value));
    stream.forEach(s -> {
        if (heap.size() < k) heap.add(s);
        else if (s.value() > heap.peek().value()) {
            heap.poll();                         // drop the weakest survivor
            heap.add(s);
        }
    });
    List<Score> out = new ArrayList<>(heap);     // heap order, NOT sorted
    out.sort(Comparator.comparingInt(Score::value).reversed());
    return out;                                  // O(n log k + k log k)
}
