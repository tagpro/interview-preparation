Part seven

The patterns that solve most of the rest

Below the level of named algorithms there is a smaller set of techniques that recur endlessly: keep two indices instead of two loops, precompute the sums, take the locally best option and prove it is safe, search exhaustively but prune early, and use the bits directly. None is difficult. Recognising which one a problem wants is the entire skill.

Two pointers and the sliding window

A nested loop over pairs is O(n²) because it reconsiders pairs it has already ruled out. When the data has an order — sorted values, or a window that only ever grows and shrinks in one direction — two indices that each move forward only can rule out whole ranges at once, and the whole sweep becomes O(n).

Opposite endsNeeds sorted input. Two-sum, three-sum, container-with-most-water, palindrome checks. Move whichever end can possibly improve the answer.
Same directionThe sliding window. Longest or shortest run satisfying a condition. Grow the right edge; when the condition breaks, advance the left until it holds again.
Fast and slowCycle detection, middle of a list, k-th from the end. One pointer moves twice as fast, or starts k ahead.
Two sequencesMerging, intersecting sorted sets, and diffing. Advance whichever side is behind — the merge step of merge sort is exactly this.

Snippets in Go, Python and Java — switch in the bar, or press g / p / j.

// window.go — the sliding-window template. Almost every "longest or
// shortest run such that ..." problem is this eight-line shape with a
// different condition.

func LongestUnique(s string) (best, at int) {
    last := make(map[byte]int, 64)               // byte -> last index seen
    lo := 0
    for hi := 0; hi < len(s); hi++ {
        if prev, seen := last[s[hi]]; seen && prev >= lo {
            lo = prev + 1                        // jump past the duplicate; never back
        }
        last[s[hi]] = hi
        if hi-lo+1 > best {
            best, at = hi-lo+1, lo
        }
    }
    return best, at
}

// Opposite ends, on sorted input. Each iteration eliminates one whole
// row or column of the pair matrix, which is why n steps suffice for
// n^2 pairs.
func TwoSum(sorted []int, target int) (int, int, bool) {
    i, j := 0, len(sorted)-1
    for i < j {
        switch sum := sorted[i] + sorted[j]; {
        case sum == target:
            return i, j, true
        case sum < target:
            i++                                  // no pair with sorted[i] can work
        default:
            j--                                  // no pair with sorted[j] can work
        }
    }
    return 0, 0, false
}

// In-place filter: read and write pointers over one array. This is the
// same idea and it is how every "remove duplicates from a sorted array
// without allocating" is written.
func Compact(xs []int) []int {
    w := 0
    for r := range xs {
        if r == 0 || xs[r] != xs[r-1] {
            xs[w] = xs[r]
            w++
        }
    }
    return xs[:w]                                // slices.Compact does exactly this
}

from collections import Counter

def longest_unique(s: str) -> str:
    """Longest substring with no repeated character. O(n)."""
    last: dict[str, int] = {}
    lo = best = at = 0
    for hi, ch in enumerate(s):
        if (prev := last.get(ch, -1)) >= lo:
            lo = prev + 1                        # the left edge only ever advances
        last[ch] = hi
        if hi - lo + 1 > best:
            best, at = hi - lo + 1, lo
    return s[at:at + best]

def min_window_covering(s: str, need: str) -> str:
    """Shortest substring containing every character of `need`. O(n).

    The 'shortest' variant grows the right edge until the window is
    valid, then shrinks the left as far as it can while staying valid.
    Both edges still only move forward.
    """
    want = Counter(need)
    missing = len(need)
    lo = best_lo = 0
    best = float("inf")
    for hi, ch in enumerate(s):
        if want[ch] > 0:
            missing -= 1
        want[ch] -= 1
        while missing == 0:                      # valid: try to shrink
            if hi - lo + 1 < best:
                best, best_lo = hi - lo + 1, lo
            want[s[lo]] += 1
            if want[s[lo]] > 0:
                missing += 1
            lo += 1
    return "" if best == float("inf") else s[best_lo:best_lo + best]

# Fixed-size windows do not need any of this: keep a running sum and
# subtract the element leaving as you add the one arriving.
def rolling_mean(xs: list[float], k: int) -> list[float]:
    total = sum(xs[:k])
    out = [total / k]
    for i in range(k, len(xs)):
        total += xs[i] - xs[i - k]               # O(1) per step, not O(k)
        out.append(total / k)
    return out
// Window.java — the template, with the two things Java makes you think
// about: char vs code point, and the array-as-counter trick.

static int longestUnique(String s) {
    int[] last = new int[128];                   // ASCII: faster than a HashMap
    Arrays.fill(last, -1);
    int lo = 0, best = 0;
    for (int hi = 0; hi < s.length(); hi++) {
        char c = s.charAt(hi);
        if (last[c] >= lo) lo = last[c] + 1;     // jump; never move lo back
        last[c] = hi;
        best = Math.max(best, hi - lo + 1);
    }
    return best;
}

// Two pointers over two sorted lists: the intersection, in O(n + m)
// rather than the O(n*m) of contains() in a loop.
static List<Integer> intersect(int[] a, int[] b) {
    List<Integer> out = new ArrayList<>();
    int i = 0, j = 0;
    while (i < a.length && j < b.length) {
        if (a[i] == b[j]) { out.add(a[i]); i++; j++; }
        else if (a[i] < b[j]) i++;               // advance whichever is behind
        else j++;
    }
    return out;
}

// The read/write pointer, in place:
static int compact(int[] xs) {
    int w = 0;
    for (int r = 0; r < xs.length; r++)
        if (r == 0 || xs[r] != xs[r - 1]) xs[w++] = xs[r];
    return w;                                    // the new logical length
}

Prefix sums and difference arrays

If you are going to be asked many range questions about an array that is not changing, answer them all at once first. A prefix-sum array turns “the sum from i to j” from an O(n) loop into a single subtraction — and its mirror image, the difference array, turns “add 5 to everything from i to j” into two writes.

// prefix.go — build once in O(n), answer every range query in O(1).

func Prefix(xs []int) []int {
    p := make([]int, len(xs)+1)                  // note the +1: p[0] = 0 removes
    for i, v := range xs {                       // every "if i == 0" branch below
        p[i+1] = p[i] + v
    }
    return p
}

sum := p[j+1] - p[i]                             // inclusive range [i, j], O(1)

// The difference array is the same idea run backwards: to add `delta`
// to every element in [i, j], record it at the two ENDS and integrate
// once at the end. k range updates in O(k + n) instead of O(kn).
func ApplyRanges(n int, updates [][3]int) []int {
    diff := make([]int, n+1)
    for _, u := range updates {
        lo, hi, delta := u[0], u[1], u[2]
        diff[lo] += delta
        diff[hi+1] -= delta                      // the +1 slot is why diff is n+1
    }
    out := make([]int, n)
    running := 0
    for i := range out {
        running += diff[i]
        out[i] = running
    }
    return out
}

// This is how "how many meetings overlap at the busiest moment" is
// solved: +1 at each start, -1 at each end, then a running maximum of
// the integral. No interval tree required.

// When the array DOES change between queries, a prefix sum is wrong --
// every update invalidates the whole tail. That is what a Fenwick tree
// (binary indexed tree) is for: O(log n) update and O(log n) prefix
// query, in n words of memory and about fifteen lines.

running_max = list(accumulate(xs, max))

# The prefix-sum + hash-map combination, which is the answer to a
# surprising number of questions: "how many subarrays sum to k?"
# A subarray (i, j] sums to k exactly when p[j] - p[i] == k, so for
# each j we need the count of earlier prefixes equal to p[j] - k.
def count_subarrays_summing_to(xs: list[int], k: int) -> int:
    from collections import defaultdict
    seen = defaultdict(int)
    seen[0] = 1                                  # the empty prefix
    running = total = 0
    for v in xs:
        running += v
        total += seen[running - k]               # O(1) per element
        seen[running] += 1
    return total

# O(n) and one pass. The obvious version is O(n^2), and the difference
# is entirely "remember what you have already computed".

# Two dimensions: a 2-D prefix sum answers "sum of this rectangle" in
# four lookups by inclusion-exclusion.
#   area = P[r2][c2] - P[r1][c2] - P[r2][c1] + P[r1][c1]
// Prefix.java — and the Fenwick tree, for when the array changes.

static long[] prefix(int[] xs) {
    long[] p = new long[xs.length + 1];          // long: prefix sums overflow int
    for (int i = 0; i < xs.length; i++) p[i + 1] = p[i] + xs[i];
    return p;
}

// A Fenwick tree: prefix sums that survive updates. Each slot i holds
// the sum of a block whose length is the lowest set bit of i, so both
// operations walk O(log n) slots. Fifteen lines, and it replaces a
// segment tree for anything that only needs sums.
final class Fenwick {
    private final long[] t;

    Fenwick(int n) { t = new long[n + 1]; }      // 1-indexed

    void add(int i, long delta) {                // O(log n)
        for (i++; i < t.length; i += i & -i) t[i] += delta;
    }

    long prefix(int i) {                         // sum of [0, i), O(log n)
        long s = 0;
        for (; i > 0; i -= i & -i) s += t[i];
        return s;
    }

    long range(int lo, int hi) { return prefix(hi) - prefix(lo); }
}

// `i & -i` isolates the lowest set bit -- the one bit trick worth
// knowing by name, and the whole reason this structure is so small.
//
// Fenwick handles sums (and anything invertible). For minimum, maximum
// or gcd over a range, you need a segment tree, which is larger but
// handles any associative operation.

Greedy, and how to know it is safe

A greedy algorithm takes the best-looking option at each step and never reconsiders. When it works it is the simplest and fastest thing available; when it does not, it produces a plausible wrong answer with no error. The difference is not intuition — it is a proof, and there are two standard ones.

ProblemGreedy ruleSafe?
Interval scheduling — most non-overlapping meetingsAlways take the one that ends earliestYes. Exchange argument: swapping any optimal solution’s first meeting for the earliest-ending one leaves it valid and no smaller.
Coin change, UK or US coinsTake the largest coin that fitsYes, for those coin systems. No in general: with coins 1, 3, 4 making 6, greedy gives 4+1+1 and the answer is 3+3.
Fractional knapsackTake the highest value-per-weight firstYes. You can take a fraction of the last item, so there is no wasted capacity.
0/1 knapsackSame ruleNo. Indivisibility breaks it, and this is exactly why the DP in part six exists.
Minimum spanning treeTake the lightest edge that joins two componentsYes. The cut property, proved in part five.
Huffman codingRepeatedly merge the two least frequent symbolsYes. Exchange argument on the two deepest leaves.

The two proofs, in one sentence each

The exchange argument: take any optimal solution, show you can swap its first choice for the greedy choice without making it worse, then repeat — so a solution that is greedy all the way down is optimal too. The matroid / cut argument: show that the locally best element is in some optimal solution, and that removing it leaves a problem of the same kind. If you cannot produce one of these two for your rule, assume greedy is wrong and look for a counterexample; they are usually small, and finding one takes less time than the incident does.

// greedy.go — interval scheduling. The whole algorithm is the sort key,
// and choosing the wrong key is the classic error.

type Interval struct{ Start, End int }

// Sort by END time, not by start and not by duration. A meeting that
// finishes early leaves the most room for everything after it, and that
// sentence is the proof.
func MaxNonOverlapping(ivs []Interval) []Interval {
    slices.SortFunc(ivs, func(a, b Interval) int { return cmp.Compare(a.End, b.End) })

    out := make([]Interval, 0, len(ivs))
    lastEnd := math.MinInt
    for _, iv := range ivs {
        if iv.Start >= lastEnd {                 // no overlap with the last kept
            out = append(out, iv)
            lastEnd = iv.End
        }
    }
    return out
}

// Merging overlapping intervals: sort by START, then extend or emit.
// Different question, different key -- and that is the point.
func Merge(ivs []Interval) []Interval {
    slices.SortFunc(ivs, func(a, b Interval) int { return cmp.Compare(a.Start, b.Start) })

    out := ivs[:0]                               // reuse the backing array
    for _, iv := range ivs {
        if n := len(out); n > 0 && iv.Start <= out[n-1].End {
            out[n-1].End = max(out[n-1].End, iv.End)
        } else {
            out = append(out, iv)
        }
    }
    return out
}

// is ordered, the sweep is a single pass with one variable of state.
# greedy.py — Huffman coding, which is greedy plus a heap and is the
# clearest example of a rule that is provably safe.
import heapq
from collections import Counter
from itertools import count

def huffman(text: str) -> dict[str, str]:
    """Optimal prefix-free codes: frequent symbols get shorter codes."""
    freq = Counter(text)
    if len(freq) < 2:
        return {ch: "0" for ch in freq}

    seq = count()                                # tie-breaker: never compare dicts
    heap = [(n, next(seq), {ch: ""}) for ch, n in freq.items()]
    heapq.heapify(heap)

    while len(heap) > 1:
        # The greedy step: the two least frequent symbols must be the
        # two DEEPEST leaves in some optimal tree, so merging them now
        # cannot cost anything.
        n1, _, a = heapq.heappop(heap)
        n2, _, b = heapq.heappop(heap)
        merged = {ch: "0" + code for ch, code in a.items()}
        merged |= {ch: "1" + code for ch, code in b.items()}
        heapq.heappush(heap, (n1 + n2, next(seq), merged))

    return heap[0][2]

# A counterexample to greedy coin change, small enough to keep in your
# head as a reminder that "obviously optimal" needs a proof:
#     coins = [1, 3, 4], target = 6
#     greedy -> 4 + 1 + 1   (three coins)
#     optimal -> 3 + 3      (two)
// Greedy.java — the meeting-rooms family, which is really a sweep line.

record Interval(int start, int end) {}

// "How many rooms do I need?" is not an interval-scheduling problem: it
// is the maximum overlap, and the neat solution is a heap of end times.
static int roomsNeeded(List<Interval> meetings) {
    List<Interval> sorted = new ArrayList<>(meetings);
    sorted.sort(Comparator.comparingInt(Interval::start));

    // The heap holds the end time of every meeting still running. Its
    // size is the number of rooms in use; its root is the one that
    // frees up first.
    PriorityQueue<Integer> running = new PriorityQueue<>();
    for (Interval m : sorted) {
        if (!running.isEmpty() && running.peek() <= m.start()) {
            running.poll();                      // a room came free: reuse it
        }
        running.add(m.end());
    }
    return running.size();                       // the peak, O(n log n)
}

// The same answer as a difference array, in O(n log n) but with no heap
// and a much smaller constant: +1 at every start, -1 at every end, sort
// the events, and take the running maximum. Pick whichever you find
// easier to get right at 3 a.m.; they are the same algorithm seen from
// two directions.

Backtracking: exhaustive search that stops early

Some problems have no better answer than trying everything. Backtracking makes “everything” tractable by building candidates one decision at a time and abandoning a branch the moment it cannot possibly lead anywhere — so the pruning, not the enumeration, is the algorithm.

// backtrack.go — the template. Choose, recurse, un-choose. That last
// step is what makes it backtracking rather than brute force with
// copies, and it keeps the allocation count at zero.

func Permutations(xs []int) [][]int {
    var out [][]int
    var cur []int
    used := make([]bool, len(xs))

    var walk func()
    walk = func() {
        if len(cur) == len(xs) {
            out = append(out, slices.Clone(cur)) // CLONE: cur keeps mutating
            return
        }
        for i, v := range xs {
            if used[i] {
                continue
            }
            used[i], cur = true, append(cur, v)  // choose
            walk()                               // explore
            used[i], cur = false, cur[:len(cur)-1] // un-choose
        }
    }
    walk()
    return out
}

// N-queens: the same template, with the pruning that makes it possible.
// Three bitmasks track which column and which of the two diagonals are
// already attacked, so "is this square safe" is one AND.
func NQueens(n int) int {
    var count int
    var place func(row, cols, diag, anti int)
    place = func(row, cols, diag, anti int) {
        if row == n {
            count++
            return
        }
        // free = the squares in this row attacked by nothing
        free := ^(cols | diag | anti) & (1<<n - 1)
        for free != 0 {
            bit := free & -free                  // lowest set bit: one square
            free ^= bit
            place(row+1, cols|bit, (diag|bit)<<1, (anti|bit)>>1)
        }
    }
    place(0, 0, 0, 0)
    return count
}

# Reach for itertools first: it is C, lazy, and correct. Write the
# recursion only when you need to PRUNE, which the library cannot do.

def subsets_summing_to(xs: list[int], target: int) -> list[list[int]]:
    """All subsets with a given sum, pruning branches that cannot reach it."""
    xs = sorted(xs, reverse=True)                # big first: prune sooner
    suffix = [0] * (len(xs) + 1)                 # suffix[i] = sum of xs[i:]
    for i in range(len(xs) - 1, -1, -1):
        suffix[i] = suffix[i + 1] + xs[i]

    out: list[list[int]] = []
    cur: list[int] = []

    def walk(i: int, remaining: int) -> None:
        if remaining == 0:
            out.append(cur.copy())               # copy: cur keeps mutating
            return
        # The two prunes that turn 2^n into something finishable:
        if i == len(xs) or remaining < 0:
            return
        if suffix[i] < remaining:                # everything left is not enough
            return
        cur.append(xs[i])                        # choose
        walk(i + 1, remaining - xs[i])
        cur.pop()                                # un-choose
        walk(i + 1, remaining)                   # skip this element

    walk(0, target)
    return out
// Backtrack.java — a Sudoku solver, which is the pruning argument in
// its most familiar form.

static boolean solve(char[][] board) {
    for (int r = 0; r < 9; r++) {
        for (int c = 0; c < 9; c++) {
            if (board[r][c] != '.') continue;
            for (char v = '1'; v <= '9'; v++) {
                if (!isValid(board, r, c, v)) continue;
                board[r][c] = v;                 // choose
                if (solve(board)) return true;   // explore
                board[r][c] = '.';               // un-choose
            }
            return false;                        // no digit fits: this branch is dead
        }
    }
    return true;                                 // no empty cell left: solved
}

static boolean isValid(char[][] b, int r, int c, char v) {
    int br = r / 3 * 3, bc = c / 3 * 3;
    for (int i = 0; i < 9; i++)
        if (b[r][i] == v || b[i][c] == v || b[br + i / 3][bc + i % 3] == v)
            return false;
    return true;
}

Bit manipulation, for the handful of times it matters

Bit tricks are over-taught as puzzles and under-taught as what they actually are: a set of up to 64 elements in one register, with union, intersection and membership as single instructions. That framing is what makes them useful in real code — permission masks, feature flags, small-set dynamic programming, and the innermost loop of a search.

// step per PRESENT element, not per possible one.
type Set uint64

func (s Set) Add(i int) Set      { return s | 1<<i }
func (s Set) Has(i int) bool     { return s&(1<<i) != 0 }
func (s Set) Union(o Set) Set    { return s | o }
func (s Set) Inter(o Set) Set    { return s & o }

func (s Set) Each(f func(int)) {
    for s != 0 {
        i := bits.TrailingZeros64(uint64(s))
        f(i)

    }
}
