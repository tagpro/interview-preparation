Part six

Recursion and dynamic programming

Dynamic programming has a reputation for being hard that it does not deserve. It is one idea — a recursive definition whose subproblems repeat, computed once each instead of once per occurrence — and the difficulty is never the technique. It is finding the recurrence, and that is a matter of naming the state precisely enough that the answer for a state depends only on smaller states.

From exponential recursion to a table, in three steps

Every dynamic program starts as an honest recursive definition that is unusably slow, because it recomputes the same subproblem an exponential number of times. Memoising it is a mechanical change that fixes the complexity without touching the logic. Turning it into a loop is a second, also mechanical, change that removes the stack.

Snippets in Go, Python and Java — switch in the bar, or press g / p / j.

// dp.go — Go has no memoisation decorator, so the cache is explicit.
// That is not a loss: it makes the state space visible, which is the
// thing you are supposed to be thinking about.

// Step 1: the honest recursion. O(2^n), and correct.
func coinsNaive(amount int, coins []int) int {
    if amount == 0 {
        return 0
    }
    best := math.MaxInt
    for _, c := range coins {
        if c <= amount {
            if sub := coinsNaive(amount-c, coins); sub != math.MaxInt {
                best = min(best, sub+1)
            }
        }
    }
    return best
}

// Step 2: memoise. The state is a single int, so the cache is a slice.
// O(amount * len(coins)).
func coinsMemo(amount int, coins []int) int {
    memo := make([]int, amount+1)
    for i := range memo {
        memo[i] = -1                             // -1 = not computed yet
    }
    var solve func(int) int
    solve = func(a int) int {
        if a == 0 {
            return 0
        }
        if memo[a] != -1 {
            return memo[a]
        }
        best := math.MaxInt
        for _, c := range coins {
            if c <= a {
                if sub := solve(a - c); sub != math.MaxInt {
                    best = min(best, sub+1)
                }
            }
        }
        memo[a] = best
        return best
    }
    return solve(amount)
}

// Step 3: bottom-up. Same complexity, no recursion, no stack, and the
// loop order IS the dependency order -- every a-c is smaller than a.
func coins(amount int, cs []int) int {
    dp := make([]int, amount+1)
    for a := 1; a <= amount; a++ {
        dp[a] = math.MaxInt
        for _, c := range cs {
            if c <= a && dp[a-c] != math.MaxInt {
                dp[a] = min(dp[a], dp[a-c]+1)
            }
        }
    }
    return dp[amount]
}
# dp.py — in Python step 2 is one line, which makes it the fastest
# language to explore a recurrence in.
from functools import cache

# Step 1 and 2 together. The decorator IS the memoisation: it keys the
# cache on the argument tuple, so the arguments must be hashable.
@cache
def coins(amount: int, coins_t: tuple[int, ...]) -> float:
    if amount == 0:
        return 0
    return min((coins(amount - c, coins_t) + 1
                for c in coins_t if c <= amount), default=float("inf"))

# Two rules for @cache on a recursive function:
#   - every argument must be hashable, so pass tuples, not lists
#   - the cache is global and unbounded; it outlives the call. Use
#     lru_cache(maxsize=...) or clear it if the inputs are unbounded.

# Step 3, when the recursion depth would exceed 1000 or the constant
# factor matters:
def coins_iter(amount: int, cs: list[int]) -> float:
    dp = [0] + [float("inf")] * amount
    for a in range(1, amount + 1):
        dp[a] = min((dp[a - c] + 1 for c in cs if c <= a), default=float("inf"))
    return dp[amount]

# Reconstructing the ANSWER, not just its cost: keep the choice.
def coins_which(amount: int, cs: list[int]) -> list[int] | None:
    dp = [0] + [None] * amount                   # dp[a] = coin used to reach a
    reach = [0] + [False] * amount
    for a in range(1, amount + 1):
        for c in cs:
            if c <= a and reach[a - c]:
                dp[a], reach[a] = c, True
                break
    if not reach[amount]:
        return None
    out = []
    while amount:                                # walk the choices backwards
        out.append(dp[amount])
        amount -= dp[amount]
    return out
// Dp.java — an int[] cache and a sentinel. Java has no memo decorator
// either; computeIfAbsent on a Map works but boxes every key.

static final int UNSET = -1, IMPOSSIBLE = Integer.MAX_VALUE;

// Step 2: top-down with an explicit cache.
static int coinsMemo(int amount, int[] coins, int[] memo) {
    if (amount == 0) return 0;
    if (memo[amount] != UNSET) return memo[amount];

    int best = IMPOSSIBLE;
    for (int c : coins) {
        if (c <= amount) {
            int sub = coinsMemo(amount - c, coins, memo);
            if (sub != IMPOSSIBLE) best = Math.min(best, sub + 1);
        }
    }
    return memo[amount] = best;
}

// Step 3: bottom-up. Prefer this in Java -- the default stack is around
// 512 KB and a recursion of depth 100,000 will not survive it.
static int coins(int amount, int[] cs) {
    int[] dp = new int[amount + 1];
    Arrays.fill(dp, IMPOSSIBLE);
    dp[0] = 0;
    for (int a = 1; a <= amount; a++)
        for (int c : cs)
            if (c <= a && dp[a - c] != IMPOSSIBLE)
                dp[a] = Math.min(dp[a], dp[a - c] + 1);
    return dp[amount];
}

// Note the guard `dp[a - c] != IMPOSSIBLE` in both. Without it,
// IMPOSSIBLE + 1 overflows to Integer.MIN_VALUE and min() cheerfully
// picks it -- an "unreachable amount costs minus two billion coins" bug
// that produces a plausible-looking wrong answer rather than a crash.

Two dimensions: the table, and reading the answer back out

Edit distance is the archetype for two-dimensional dynamic programming, and it is worth knowing in full because diff, spell checkers, fuzzy search, DNA alignment and merge algorithms are all variations of it. Each cell asks one question: given the best answers for three smaller problems, what is the best answer here?

// edit.go — the rolling-row version, which is what you ship when you
// only need the distance. Two rows instead of m*n.

func EditDistance(a, b string) int {
    ra, rb := []rune(a), []rune(b)               // runes: "café" is 4, not 5
    if len(ra) < len(rb) {
        ra, rb = rb, ra                          // keep the short side as the row
    }

    prev := make([]int, len(rb)+1)
    curr := make([]int, len(rb)+1)
    for j := range prev {
        prev[j] = j                              // turning "" into b[:j] costs j
    }

    for i := 1; i <= len(ra); i++ {
        curr[0] = i                              // turning a[:i] into "" costs i
        for j := 1; j <= len(rb); j++ {
            sub := prev[j-1]
            if ra[i-1] != rb[j-1] {
                sub++
            }
            curr[j] = min(
                prev[j]+1,                       // delete a[i-1]
                curr[j-1]+1,                     // insert b[j-1]
                sub,                             // substitute, or free on a match
            )
        }
        prev, curr = curr, prev                  // swap, do not allocate
    }
    return prev[len(rb)]
}

// Early exit for a fuzzy-search threshold: if the smallest value in a
// whole row already exceeds the limit, every later row is worse, so you
// can stop. On a dictionary scan this is the difference between usable
// and not.
//
// Longest common subsequence is the same table with a different
// recurrence -- max instead of min, and only the diagonal advances on a
// match. That is what `diff` computes, then reads back as the edit script.
# edit.py — the full table, because the backtrace is the interesting
# half and the reason `diff` output looks the way it does.

def edit_table(a: str, b: str) -> list[list[int]]:
    n, m = len(a), len(b)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dp[i][0] = i                             # delete everything
    for j in range(m + 1):
        dp[0][j] = j                             # insert everything
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            dp[i][j] = min(
                dp[i - 1][j] + 1,                # delete a[i-1]
                dp[i][j - 1] + 1,                # insert b[j-1]
                dp[i - 1][j - 1] + (a[i - 1] != b[j - 1]),   # substitute/match
            )
    return dp

def edit_script(a: str, b: str) -> list[str]:
    """Walk backwards from the corner, taking whichever choice produced it."""
    dp = edit_table(a, b)
    i, j, out = len(a), len(b), []
    while i > 0 or j > 0:
        if i > 0 and dp[i][j] == dp[i - 1][j] + 1:
            out.append(f"-{a[i - 1]}"); i -= 1
        elif j > 0 and dp[i][j] == dp[i][j - 1] + 1:
            out.append(f"+{b[j - 1]}"); j -= 1
        else:
            if a[i - 1] != b[j - 1]:
                out.append(f"~{a[i - 1]}{b[j - 1]}")
            i -= 1; j -= 1
    return out[::-1]

# edit_script("kitten", "sitting") -> ['~ks', '~ie', '+g'] with the
# matched letters skipped: three edits, which is the distance.
#
# difflib.SequenceMatcher in the standard library solves the related
# problem (longest matching blocks) with a different algorithm tuned for
# human-readable diffs rather than minimal edit count.
// Edit.java — one int[] row, which is all the distance needs.

static int editDistance(String a, String b) {
    if (a.length() < b.length()) { String t = a; a = b; b = t; }

    int[] prev = new int[b.length() + 1];
    int[] curr = new int[b.length() + 1];
    for (int j = 0; j <= b.length(); j++) prev[j] = j;

    for (int i = 1; i <= a.length(); i++) {
        curr[0] = i;
        for (int j = 1; j <= b.length(); j++) {
            int sub = prev[j - 1] + (a.charAt(i - 1) == b.charAt(j - 1) ? 0 : 1);
            curr[j] = Math.min(Math.min(prev[j] + 1, curr[j - 1] + 1), sub);
        }
        int[] swap = prev; prev = curr; curr = swap;   // reuse both arrays
    }
    return prev[b.length()];
}

The recurrences worth memorising

There are perhaps six dynamic programs that between them cover most of what you will meet. They are worth knowing not as code to recall but as state definitions — once you can say precisely what dp[i][j] means, the recurrence usually writes itself and the base cases become obvious.

ProblemStateRecurrenceCost
0/1 knapsackdp[i][w] = best value using the first i items within weight wmax(skip item i, take it and add dp[i-1][w-wi])O(nW)
Coin changedp[a] = fewest coins making amount a1 + min over coins of dp[a-c]O(aC)
Longest common subsequencedp[i][j] = LCS of the two prefixesmatch ? dp[i-1][j-1]+1 : max(dp[i-1][j], dp[i][j-1])O(nm)
Longest increasing subsequencetails[k] = smallest tail of any increasing run of length k+1Binary search for the position to replace — the greedy-plus-bisect trickO(n log n)
Maximum subarray (Kadane)best = largest sum of a run ending exactly heremax(this element, this element + previous best)O(n)
Word breakdp[i] = can the first i characters be split into wordsOR over j of (dp[j] and s[j:i] is a word)O(n²)
 
// classics.go — Kadane and LIS, the two whose clever versions beat the
// obvious table outright.

// Kadane: maximum sum of any contiguous run, in one pass. The state is
// "best run ENDING HERE", which either extends the previous run or
// starts fresh -- and that choice is the whole algorithm.
func MaxSubarray(xs []int) (best, lo, hi int) {
    best, cur, start := math.MinInt, 0, 0
    for i, v := range xs {
        if cur <= 0 {
            cur, start = v, i                    // the past is a liability: drop it
        } else {
            cur += v
        }
        if cur > best {
            best, lo, hi = cur, start, i
        }
    }
    return best, lo, hi
}

// Longest increasing subsequence in O(n log n). tails[k] holds the
// smallest possible tail of an increasing subsequence of length k+1;
// it is always sorted, so the position is a binary search.
func LIS(xs []int) int {
    tails := make([]int, 0, len(xs))
    for _, v := range xs {
        i, _ := slices.BinarySearch(tails, v)
        if i == len(tails) {
            tails = append(tails, v)             // v extends the longest run
        } else {
            tails[i] = v                         // v gives length i+1 a better tail
        }
    }
    return len(tails)                            // NOTE: the length, not the run
}

// `tails` is not itself a valid subsequence -- a common misreading. To
// recover the actual run, record a predecessor index for each element
// as you place it, then walk backwards from the last one appended.
# classics.py — knapsack, and the space optimisation that catches
# people out.

def knapsack(weights: list[int], values: list[int], cap: int) -> int:
    """0/1 knapsack: each item may be taken at most once."""
    dp = [0] * (cap + 1)                         # dp[w] = best value within w
    for wt, val in zip(weights, values):
        # BACKWARDS. Forwards would let this item be picked twice,
        # because dp[w - wt] would already include it. That single
        # reversal is the difference between 0/1 and unbounded knapsack.
        for w in range(cap, wt - 1, -1):
            dp[w] = max(dp[w], dp[w - wt] + val)
    return dp[cap]

def unbounded_knapsack(weights, values, cap) -> int:
    dp = [0] * (cap + 1)
    for wt, val in zip(weights, values):
        for w in range(wt, cap + 1):             # forwards: reuse is allowed
            dp[w] = max(dp[w], dp[w - wt] + val)
    return dp[cap]

# O(nW) is "pseudo-polynomial": W is a VALUE, not an input length, so a
# capacity of 10^9 makes this table impossible even with ten items.
# Knapsack is NP-hard; the DP is only fast because W is usually small.

def lis_with_run(xs: list[int]) -> list[int]:
    """The actual subsequence, not just its length."""
    import bisect
    tails, tail_idx, prev = [], [], [-1] * len(xs)
    for i, v in enumerate(xs):
        k = bisect.bisect_left(tails, v)
        if k == len(tails):
            tails.append(v); tail_idx.append(i)
        else:
            tails[k], tail_idx[k] = v, i
        prev[i] = tail_idx[k - 1] if k else -1
    out, i = [], tail_idx[-1] if tail_idx else -1
    while i >= 0:
        out.append(xs[i]); i = prev[i]
    return out[::-1]
// Classics.java — word break, which is the one that most often shows up
// disguised as something else (tokenising, URL routing, segmentation).

static boolean wordBreak(String s, Set<String> dict) {
    int n = s.length();
    boolean[] dp = new boolean[n + 1];
    dp[0] = true;                                // the empty prefix is splittable

    int longest = dict.stream().mapToInt(String::length).max().orElse(0);

    for (int i = 1; i <= n; i++) {
        // Only look back as far as the longest word: turns O(n^2) into
        // O(n * longest), which for real dictionaries is linear.
        for (int j = Math.max(0, i - longest); j < i; j++) {
            if (dp[j] && dict.contains(s.substring(j, i))) {
                dp[i] = true;
                break;
            }
        }
    }
    return dp[n];
}

// substring() is O(length) and allocates, so this does O(n^2) character
// copying on top of the O(n^2) loop. A trie over the dictionary removes
// both: walk forward from j one character at a time, and stop the
// instant the prefix leaves the trie.

static int maxSubarray(int[] xs) {               // Kadane, O(n), O(1)
    int best = Integer.MIN_VALUE, cur = 0;
    for (int v : xs) {
        cur = Math.max(v, cur + v);              // extend, or start again here
        best = Math.max(best, cur);
    }
    return best;
}
