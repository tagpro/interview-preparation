Part three

Trees, heaps and the structures built on them

A tree is what you reach for when a hash table gives you the wrong thing: order. Hash lookup is faster, but it cannot answer “the next key after this one”, “everything between these two” or “the smallest thing left”. Those three questions are what this part is about, plus the two specialised trees — the trie and the disjoint-set forest — that solve problems nothing else solves cleanly.

Binary trees, and the four orders you can walk them in

A binary tree is a node with up to two children. Everything interesting about it comes from the order in which you visit the nodes — and the three recursive orders differ by one line moving, which is the clearest illustration in the subject that an algorithm is a sequence, not a shape.

Pre-orderRoot, left, right. Serialising a tree, copying it, or any task where the parent must be handled before its children — rendering nested UI, applying inherited permissions.
In-orderLeft, root, right. On a binary search tree this yields sorted order, which is the property every sorted map is built on.
Post-orderLeft, right, root. Freeing, computing a value from children (subtree sizes, directory sizes), and evaluating an expression tree.
Level-orderBy depth, using a queue. Finding the shallowest anything, printing a tree by rows, and any “fewest steps” question.

// tree.go — the three orders, then the two things you actually compute.

type TreeNode struct {
    Val         int
    Left, Right *TreeNode
}

func Walk(n *TreeNode, visit func(int)) {
    if n == nil {
        return
    }
    // visit(n.Val)          <- pre-order goes here
    Walk(n.Left, visit)
    visit(n.Val)                                 // in-order
    Walk(n.Right, visit)
    // visit(n.Val)          <- post-order goes here
}

// Level order: a queue, and one loop per level so you know where the
// rows break. That inner "for range len(q)" is the whole trick.
func Levels(root *TreeNode) [][]int {
    if root == nil {
        return nil
    }
    var out [][]int
    q := []*TreeNode{root}
    for len(q) > 0 {
        row := make([]int, 0, len(q))
        next := make([]*TreeNode, 0, len(q)*2)
        for _, n := range q {                    // exactly one level
            row = append(row, n.Val)
            if n.Left != nil {
                next = append(next, n.Left)
            }
            if n.Right != nil {
                next = append(next, n.Right)
            }
        }
        out, q = append(out, row), next
    }
    return out
}

// Height is post-order: you cannot know yours until both children know
// theirs. Almost every "compute something about a subtree" is this shape.
func Height(n *TreeNode) int {
    if n == nil {
        return 0
    }
    return 1 + max(Height(n.Left), Height(n.Right))
}
# tree.py — a generator makes the traversal lazy, which matters when
# the caller wants the first match rather than all of them.
from collections import deque
from typing import Iterator, Optional

class Node:
    __slots__ = ("val", "left", "right")         # no __dict__: ~40% less memory

    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right

def in_order(n: Optional[Node]) -> Iterator[int]:
    if n is None:
        return
    yield from in_order(n.left)
    yield n.val                                  # move this line for pre/post
    yield from in_order(n.right)

def levels(root: Optional[Node]) -> list[list[int]]:
    if root is None:
        return []
    out, q = [], deque([root])
    while q:
        row = []
        for _ in range(len(q)):                  # snapshot the level's width
            n = q.popleft()
            row.append(n.val)
            if n.left:
                q.append(n.left)
            if n.right:
                q.append(n.right)
        out.append(row)
    return out

# The generator version composes with everything: itertools.islice for
# the first k, next() for the first match, zip() to compare two trees.
first_over_100 = next((v for v in in_order(root) if v > 100), None)

// Trees.java — records make an immutable tree pleasant, and sealed
// interfaces let the compiler check that you handled every shape.

sealed interface Tree permits Leaf, Branch {}
record Leaf() implements Tree {}
record Branch(int val, Tree left, Tree right) implements Tree {}

// Pattern matching in a switch: exhaustive, so a new variant is a
// compile error rather than a runtime surprise.
static int height(Tree t) {
    return switch (t) {
        case Leaf ignored -> 0;
        case Branch b -> 1 + Math.max(height(b.left()), height(b.right()));
    };
}

static void inOrder(Tree t, IntConsumer visit) {
    if (t instanceof Branch b) {
        inOrder(b.left(), visit);
        visit.accept(b.val());                   // move for pre/post order
        inOrder(b.right(), visit);
    }
}

static List<List<Integer>> levels(TreeNode root) {
    List<List<Integer>> out = new ArrayList<>();
    if (root == null) return out;
    Deque<TreeNode> q = new ArrayDeque<>(List.of(root));
    while (!q.isEmpty()) {
        int width = q.size();                    // snapshot before adding children
        List<Integer> row = new ArrayList<>(width);
        for (int i = 0; i < width; i++) {
            TreeNode n = q.poll();
            row.add(n.val);
            if (n.left != null) q.add(n.left);
            if (n.right != null) q.add(n.right);
        }
        out.add(row);
    }
    return out;
}

Search trees, balance, and why you will use somebody else’s

A binary search tree keeps everything smaller on the left and everything larger on the right, which makes lookup a walk down one path — O(log n) if the tree is balanced and O(n) if it is not. Since the shape depends entirely on insertion order, and insertion order is frequently sorted, an unbalanced BST is not an edge case. It is the default outcome.

type Index struct {
    keys []string                                // kept sorted
    vals []Record                                // parallel
}

func (ix *Index) Get(k string) (Record, bool) {
    i, ok := slices.BinarySearch(ix.keys, k)     // O(log n)
    if !ok {
        return Record{}, false
    }
    return ix.vals[i], true
}

// The insertion point is the answer to "the first key >= k" -- the same
// lower_bound that TreeMap.ceilingKey gives you.
func (ix *Index) Range(lo, hi string) []Record {
    a, _ := slices.BinarySearch(ix.keys, lo)     // first >= lo
    b, _ := slices.BinarySearch(ix.keys, hi)     // first >= hi
    return ix.vals[a:b]
}

func (ix *Index) Put(k string, v Record) {
    i, ok := slices.BinarySearch(ix.keys, k)
    if ok {
        ix.vals[i] = v
        return
    }
    ix.keys = slices.Insert(ix.keys, i, k)       // O(n) memmove -- fast per
    ix.vals = slices.Insert(ix.vals, i, v)       // byte, but O(n)
}

Why databases use B-trees and not binary trees

A balanced binary tree over a billion keys is thirty levels deep, and if the tree is on disk that is thirty random reads — roughly a millisecond and a half at NVMe latencies, several hundred milliseconds on a spinning disk. A B-tree stores hundreds of keys per node, sized to a page, so the same billion keys are four levels deep and a lookup is four reads. The asymptotics are identical (both are O(log n); only the base of the logarithm changed) and the practical difference is a factor of eight. This is the memory hierarchy from part one deciding a data structure outright, and it is why every database index is a B-tree or an LSM tree.

Heaps: the structure that only answers one question

A binary heap gives you the minimum (or maximum) in O(1) and lets you remove it or add a new element in O(log n). It cannot search, it cannot iterate in order, and it has no notion of position — and by giving all that up it becomes a plain array with no pointers at all.

Top-kKeep a heap of size k holding the best k so far, and push-then-pop for each new element. O(n log k) time and O(k) space, over a stream you cannot store.
Merging sorted runsOne heap entry per run. Pop the smallest, emit it, push that run’s next value. This is how external sorts and LSM compaction work.
Schedulers and timersKeyed by deadline. The next thing to do is always the root, and inserting a new timer is O(log n).
Dijkstra and A*Keyed by distance. The priority queue is what turns the search from “try everything” into “always expand the cheapest frontier node”.
// heap.go — container/heap is an interface, not a container: you supply
// the storage and the comparison, it supplies the sift logic.

type Job struct {
    Priority int
    Name     string
    index    int                                 // maintained by heap.Fix
}

type PQ []*Job

func (q PQ) Len() int            { return len(q) }
func (q PQ) Less(i, j int) bool  { return q[i].Priority < q[j].Priority }  // min-heap
func (q PQ) Swap(i, j int)       { q[i], q[j] = q[j], q[i]; q[i].index = i; q[j].index = j }
func (q *PQ) Push(x any)         { j := x.(*Job); j.index = len(*q); *q = append(*q, j) }
func (q *PQ) Pop() any {
    old := *q
    n := len(old)
    it := old[n-1]
    old[n-1] = nil                               // let the GC have it
    *q = old[:n-1]
    return it
}

pq := &PQ{}
heap.Init(pq)
heap.Push(pq, &Job{Priority: 3, Name: "compact"})
next := heap.Pop(pq).(*Job)                      // O(log n)

// The reason to keep `index` is heap.Fix: change a job's priority in
// place and re-sift from where it sits, in O(log n) instead of O(n).
// Dijkstra with decrease-key needs exactly this.
job.Priority = 1
heap.Fix(pq, job.index)

// Top-k over a stream, in O(n log k) and O(k) memory: keep the k
// LARGEST by holding a MIN-heap and evicting the root.
for _, v := range stream {
    if pq.Len() < k {
        heap.Push(pq, v)
    } else if v.Priority > (*pq)[0].Priority {
        heap.Pop(pq)
        heap.Push(pq, v)
    }
}

Tries: when the key has structure worth sharing

A hash map treats a key as an opaque blob to be digested. A trie treats it as a path, one edge per character, so that every key sharing a prefix shares the nodes for it. That buys three things a hash cannot offer at any price: prefix search, ordered iteration, and lookup time that depends on the key length rather than on how many keys there are.

// trie.go — autocomplete, routing tables and dictionary lookups.
// A map per node keeps it simple; a [26]*Node array is faster and only
// works for a known alphabet.

type Trie struct {
    kids map[rune]*Trie
    term bool                                    // a word ends here
}

func New() *Trie { return &Trie{kids: map[rune]*Trie{}} }

func (t *Trie) Insert(word string) {             // O(len(word))
    node := t
    for _, r := range word {
        next, ok := node.kids[r]
        if !ok {
            next = New()
            node.kids[r] = next
        }
        node = next
    }
    node.term = true
}

func (t *Trie) node(prefix string) *Trie {
    node := t
    for _, r := range prefix {
        next, ok := node.kids[r]
        if !ok {
            return nil
        }
        node = next
    }
    return node
}

// The operation a hash map cannot do: everything under a prefix, in
// sorted order, in time proportional to the OUTPUT rather than to the
// number of keys stored.
func (t *Trie) WithPrefix(prefix string, limit int) []string {
    start := t.node(prefix)
    if start == nil {
        return nil
    }
    var out []string
    var walk func(n *Trie, acc []rune)
    walk = func(n *Trie, acc []rune) {
        if len(out) >= limit {
            return
        }
        if n.term {
            out = append(out, prefix+string(acc))
        }
        for _, r := range slices.Sorted(maps.Keys(n.kids)) {   // sorted order
            walk(n.kids[r], append(acc, r))
        }
    }
    walk(start, nil)
    return out
}

END = object()                                   # a sentinel key meaning "word ends"

def build(words: list[str]) -> dict:
    root: dict = {}
    for w in words:
        node = root
        for ch in w:

def starts_with(root: dict, prefix: str, limit: int = 10) -> list[str]:
    node = root
    for ch in prefix:
        node = node.get(ch)
        if node is None:
            return []
    out: list[str] = []

    def walk(n: dict, acc: str) -> None:
        if len(out) >= limit:
            return
        if END in n:
            out.append(prefix + acc)
        for ch in sorted(k for k in n if k is not END):
            walk(n[ch], acc + ch)

    walk(node, "")
    return out

# When to use one, and when not to:
#   yes - autocomplete, longest-prefix routing, spell check, a word
#         list you query by prefix constantly
#   no  - "I have strings as keys". A dict is smaller and faster for
#         exact lookup; a trie costs a node per character and loses
#         badly on memory unless prefixes really are shared.
#
# For fixed dictionaries, pyahocorasick (Aho-Corasick) finds ALL of
# thousands of patterns in one pass over the text -- a trie plus failure
# links, and the right answer for keyword scanning.
// Trie.java — a HashMap per node is the readable version; an array of
// 26 children is roughly twice as fast when the alphabet is fixed.

final class Trie {
    private final Map<Character, Trie> kids = new TreeMap<>();   // sorted kids
    private boolean terminal;

    void insert(String word) {                   // O(word.length())
        Trie node = this;
        for (char c : word.toCharArray()) {
            node = node.kids.computeIfAbsent(c, k -> new Trie());
        }
        node.terminal = true;
    }

    private Trie descend(String prefix) {
        Trie node = this;
        for (char c : prefix.toCharArray()) {
            node = node.kids.get(c);
            if (node == null) return null;
        }
        return node;
    }

    List<String> withPrefix(String prefix, int limit) {
        Trie start = descend(prefix);
        List<String> out = new ArrayList<>();
        if (start != null) collect(start, new StringBuilder(prefix), out, limit);
        return out;
    }

    private static void collect(Trie n, StringBuilder acc, List<String> out, int limit) {
        if (out.size() >= limit) return;
        if (n.terminal) out.add(acc.toString());
        for (var e : n.kids.entrySet()) {        // TreeMap: sorted
            acc.append(e.getKey());
            collect(e.getValue(), acc, out, limit);
            acc.deleteCharAt(acc.length() - 1);  // backtrack, no copying
        }
    }
}

Disjoint sets: forty lines that answer “are these connected?”

Union-find maintains a partition of n elements under two operations: merge two groups, and ask which group something is in. With both of its optimisations it runs in near-constant amortised time — and it is the smallest piece of code on this page with a genuinely surprising complexity proof.

// dsu.go — two int slices, twenty lines, and it is used by Kruskal's
// MST, connected components, image segmentation and cycle detection in
// undirected graphs.

type DSU struct {
    parent []int
    size   []int
    groups int
}

func NewDSU(n int) *DSU {
    d := &DSU{parent: make([]int, n), size: make([]int, n), groups: n}
    for i := range d.parent {
        d.parent[i], d.size[i] = i, 1            // everyone is their own root
    }
    return d
}

// Path halving: point every other node at its grandparent. One pass,
// no recursion, and asymptotically as good as full compression.
func (d *DSU) Find(x int) int {
    for d.parent[x] != x {
        d.parent[x] = d.parent[d.parent[x]]
        x = d.parent[x]
    }
    return x
}

// Union by size. Returns false if they were already together, which is
// exactly the cycle test Kruskal needs.
func (d *DSU) Union(a, b int) bool {
    ra, rb := d.Find(a), d.Find(b)
    if ra == rb {
        return false
    }
    if d.size[ra] < d.size[rb] {
        ra, rb = rb, ra                          // smaller hangs under larger
    }
    d.parent[rb] = ra
    d.size[ra] += d.size[rb]
    d.groups--
    return true
}

// Note what it cannot do: there is no Split. Union-find is one-way, so
// a problem with deletions needs a different structure -- or, commonly,
// processing the events in reverse so deletions become unions.
# dsu.py — the same twenty lines. The classic use is counting connected
# components in one pass over the edges, with no graph built at all.

class DSU:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))
        self.size = [1] * n
        self.groups = n

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False                         # already connected -> a cycle
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        self.groups -= 1
        return True

dsu = DSU(n)

    dsu.union(a, b)

# Recursive find() with full compression is prettier and will blow the
# stack on a long chain, which is exactly the input that makes chains.
# Use the loop.
// Dsu.java — int arrays, no objects, no boxing. This is one of the few
// places where the "write it yourself" answer is unambiguously right:
// the JDK has no union-find.

final class Dsu {
    private final int[] parent;
    private final int[] size;
    private int groups;

    Dsu(int n) {
        parent = new int[n];
        size = new int[n];
        groups = n;
        for (int i = 0; i < n; i++) { parent[i] = i; size[i] = 1; }
    }

    int find(int x) {                            // iterative: no stack depth
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];       // path halving
            x = parent[x];
        }
        return x;
    }

    boolean union(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;
        if (size[ra] < size[rb]) { int t = ra; ra = rb; rb = t; }
        parent[rb] = ra;
        size[ra] += size[rb];
        groups--;
        return true;
    }

    int groups() { return groups; }
    boolean connected(int a, int b) { return find(a) == find(b); }
}

// Kruskal's minimum spanning tree is now four lines: sort the edges by
// weight, and keep each one whose union() returns true. The DSU is what
// makes "would this edge close a cycle?" an O(1) question.
