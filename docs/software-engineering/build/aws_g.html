CloudWatch: metrics from logs, and alarms worth waking for

CloudWatch charges for what you send it, in three different currencies — per custom metric, per gigabyte ingested, per query scanned. The embedded metric format lets you pay one of those and get two.

// EMF: a metric and its context in one stdout line. CloudWatch parses the
// _aws block and creates the metrics; every other field stays queryable in
// Logs Insights. No API call, no added latency, no throttling.
type metricDef struct {
    Name string `json:"Name"`
    Unit string `json:"Unit"`
}

type directive struct {
    Namespace  string      `json:"Namespace"`
    Dimensions [][]string  `json:"Dimensions"`
    Metrics    []metricDef `json:"Metrics"`
}

func emit(op string, d time.Duration, tenant, order, trace string) {
    rec := map[string]any{
        "_aws": map[string]any{
            "Timestamp": time.Now().UnixMilli(),
            "CloudWatchMetrics": []directive{{
                Namespace: "Orders",
                // TWO dimensions, not five: every unique combination is a
                // separate billed metric series, for ever.
                Dimensions: [][]string{{"service", "operation"}},
                Metrics:    []metricDef{{Name: "latency", Unit: "Milliseconds"}},
            }},
        },
        // Dimensions must also appear as top-level fields.
        "service":   "orders-api",
        "operation": op,
        "latency":   d.Milliseconds(),
        // High-cardinality context: searchable in Logs Insights, but NOT a
        // dimension, so not billed as thousands of metric series.
        "tenantId": tenant,
        "orderId":  order,
        "traceId":  trace,
    }
    _ = json.NewEncoder(os.Stdout).Encode(rec)
}
# EMF: a metric and its context in one stdout line. CloudWatch parses it and
# creates the metrics; every other field stays queryable in Logs Insights.
# No API call, no added latency, no throttling. Powertools writes the format.
from aws_lambda_powertools import Metrics
from aws_lambda_powertools.metrics import MetricUnit

metrics = Metrics(namespace="Orders", service="orders-api")


def place_order(tenant_id: str, order_id: str, elapsed_ms: float):
    # Dimensions: the two or three you would actually alarm on. Every unique
    # combination is a separate billed metric series, for ever.
    metrics.add_dimension(name="operation", value="place_order")
    metrics.add_metric(name="latency", unit=MetricUnit.Milliseconds, value=elapsed_ms)

    # High-cardinality context goes in metadata: searchable in Logs Insights,
    # attached to the same record, and NOT a billed metric series.
    metrics.add_metadata(key="tenantId", value=tenant_id)
    metrics.add_metadata(key="orderId", value=order_id)


# Correlate logs with traces so one click goes from alarm to request:
logger = Logger(service="orders-api")
logger.info("order placed", extra={"orderId": order_id})

Secrets, parameters and KMS

KMS never gives you a key — it gives you a key encrypted under a key, and a decrypted copy to use immediately. Everything above it, Secrets Manager included, is that trick plus a place to keep the ciphertext.

// Cache with a TTL, and refresh on demand when authentication fails.
// Both halves matter: the cache saves money, the invalidation survives rotation.
type SecretCache struct {
    mu    sync.RWMutex
    sm    *secretsmanager.Client
    ttl   time.Duration
    value string
    at    time.Time
}

func (c *SecretCache) Get(ctx context.Context, force bool) (string, error) {
    c.mu.RLock()
    if !force && c.value != "" && time.Since(c.at) < c.ttl {
        v := c.value
        c.mu.RUnlock()
        return v, nil
    }
    c.mu.RUnlock()

    c.mu.Lock()
    defer c.mu.Unlock()
    if !force && c.value != "" && time.Since(c.at) < c.ttl {
        return c.value, nil                    // another goroutine refreshed it
    }

    out, err := c.sm.GetSecretValue(ctx, &secretsmanager.GetSecretValueInput{
        SecretId:     aws.String("orders/db"),
        VersionStage: aws.String("AWSCURRENT"),   // never pin a version id
    })
    if err != nil {
        return "", err
    }
    c.value, c.at = aws.ToString(out.SecretString), time.Now()
    return c.value, nil
}

// At the call site: one retry with force=true turns a rotation from an outage
// into a single failed connection.
pw, _ := cache.Get(ctx, false)
if err := connect(ctx, pw); errors.Is(err, ErrAuth) {
    pw, _ = cache.Get(ctx, true)
    err = connect(ctx, pw)
}
# Cache with a TTL, and refresh on demand when authentication fails.
# Both halves matter: the cache saves money, the invalidation survives rotation.
import threading, time

SM = boto3.client("secretsmanager", config=CFG)


class SecretCache:
    def __init__(self, secret_id: str, ttl: float = 900.0):
        self._id, self._ttl = secret_id, ttl
        self._lock = threading.Lock()
        self._value: str | None = None
        self._at = 0.0

    def get(self, force: bool = False) -> str:
        fresh = self._value is not None and time.monotonic() - self._at < self._ttl
        if fresh and not force:
            return self._value
        with self._lock:
            # re-check: another thread may have refreshed while we waited
            fresh = self._value is not None and time.monotonic() - self._at < self._ttl
            if fresh and not force:
                return self._value
            res = SM.get_secret_value(
                SecretId=self._id,
                VersionStage="AWSCURRENT",     # never pin a version id
            )
            self._value, self._at = res["SecretString"], time.monotonic()
            return self._value


# At the call site: one retry with force=True turns a rotation from an outage
# into a single failed connection.
cache = SecretCache("orders/db")
try:
    connect(cache.get())
except AuthError:
    connect(cache.get(force=True))

Key policies come first

A KMS key’s own policy is authoritative: an IAM policy granting kms:Decrypt does nothing unless the key policy delegates to IAM in that account. This is the reason cross-account S3 reads fail with a KMS error that names a role which visibly has the permission.

Never put a secret in a task definition

Plain environment values are visible to anyone with ecs:DescribeTaskDefinition, and they live in every revision for ever. Use the secrets block, which resolves an ARN at task start using the execution role — the value never appears in the definition, the API, or the console.

The bill, and where it actually comes from

On a small-to-medium account the top of the bill is rarely compute. It is data movement, log ingestion, and things nobody deleted — all of which are invisible in the architecture diagram.

// Month over month, by service, biggest increase first. Ten seconds of API
// calls that answers "why is the bill up" better than any dashboard.
ce := costexplorer.NewFromConfig(cfg)

get := func(start, end string) (map[string]float64, error) {
    out, err := ce.GetCostAndUsage(ctx, &costexplorer.GetCostAndUsageInput{
        TimePeriod:  &types.DateInterval{Start: aws.String(start), End: aws.String(end)},
        Granularity: types.GranularityMonthly,
        Metrics:     []string{"UnblendedCost"},
        GroupBy: []types.GroupDefinition{
            {Type: types.GroupDefinitionTypeDimension, Key: aws.String("SERVICE")},
        },
    })
    if err != nil {
        return nil, err
    }
    m := map[string]float64{}
    for _, r := range out.ResultsByTime {
        for _, g := range r.Groups {
            v, _ := strconv.ParseFloat(aws.ToString(g.Metrics["UnblendedCost"].Amount), 64)
            m[g.Keys[0]] += v
        }
    }
    return m, nil
}

prev, _ := get("2026-07-01", "2026-08-01")
curr, _ := get("2026-08-01", "2026-09-01")
type row struct {
    svc           string
    amount, delta float64
}
rows := make([]row, 0, len(curr))
for svc, amount := range curr {
    rows = append(rows, row{svc, amount, amount - prev[svc]})
}
// a map has no order, so sort — otherwise "biggest increase first" is a lie
sort.Slice(rows, func(i, j int) bool { return rows[i].delta > rows[j].delta })

for _, r := range rows {
    if r.delta > 50 {                        // only what is worth a conversation
        fmt.Printf("%-42s %8.2f  (%+.2f)\n", r.svc, r.amount, r.delta)
    }
}
# Month over month, by service, biggest increase first. Ten seconds of API
# calls that answers "why is the bill up" better than any dashboard.
CE = boto3.client("ce", region_name="us-east-1")   # Cost Explorer is us-east-1


def by_service(start: str, end: str) -> dict[str, float]:
    res = CE.get_cost_and_usage(
        TimePeriod={"Start": start, "End": end},
        Granularity="MONTHLY",
        Metrics=["UnblendedCost"],
        GroupBy=[{"Type": "DIMENSION", "Key": "SERVICE"}],
    )
    out: dict[str, float] = {}
    for period in res["ResultsByTime"]:
        for g in period["Groups"]:
            out[g["Keys"][0]] = out.get(g["Keys"][0], 0.0) + float(
                g["Metrics"]["UnblendedCost"]["Amount"]
            )
    return out


prev = by_service("2026-07-01", "2026-08-01")
curr = by_service("2026-08-01", "2026-09-01")

rows = [(svc, amt, amt - prev.get(svc, 0.0)) for svc, amt in curr.items()]
for svc, amt, delta in sorted(rows, key=lambda r: -r[2]):
    if delta > 50:                     # only what is worth a conversation

Three habits worth more than any optimisation

Turn on Cost Anomaly Detection the day you open the account — it is free, and it emails you about the mistake rather than letting the invoice do it. Put a budget with an alert at 80% of what you expect to spend, so a runaway loop is a notification instead of a quarter. And run a monthly sweep for unattached volumes, forgotten snapshots, idle load balancers, empty NAT gateways and log groups with no retention. Between them these cost nothing and catch nearly everything.
